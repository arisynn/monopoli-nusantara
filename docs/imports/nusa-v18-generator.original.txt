import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { execFileSync } from 'node:child_process';

const ROOT = process.cwd();

function ensureDir(filePath) {
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
}

function writeText(relPath, content) {
  const full = path.join(ROOT, relPath);
  ensureDir(full);
  fs.writeFileSync(full, content.trimStart(), 'utf8');
  console.log('  [UPDATE] ' + relPath);
}

function sha256File(relPath) {
  const buf = fs.readFileSync(path.join(ROOT, relPath));
  return crypto.createHash('sha256').update(buf).digest('hex');
}

function toLibBotImports(labCode) {
  return labCode
    .replaceAll("'../../lib/content.mjs'", "'../../content.mjs'")
    .replaceAll("'../../lib/game.mjs'", "'../../game.mjs'")
    .replaceAll("'../../lib/bot/", "'../");
}

function listFilesRecursive(dirRel) {
  const results = [];
  const absDir = path.join(ROOT, dirRel);
  if (!fs.existsSync(absDir)) return results;
  for (const entry of fs.readdirSync(absDir, { withFileTypes: true })) {
    const rel = path.posix.join(dirRel.replaceAll('\\', '/'), entry.name);
    if (entry.isDirectory()) results.push(...listFilesRecursive(rel));
    else results.push(rel);
  }
  return results.sort();
}

/**
 * Tracks the 16-card FIFO queues for 'chance' and 'community' using ONLY
 * public log events (`view.log`) and held jail cards (`view.players[].cards`).
 */
export function analyzePublicCardCycles(view){
  const held=new Set((view.players||[]).flatMap(p=>p.cards||[]).map(c=>c.deck+':'+c.idx));
  const recentChance=[];
  const recentCommunity=[];
  const seenChance=new Set();
  const seenCommunity=new Set();

  for(const ev of (view.log||[])){
    const c=ev?.card||(ev?.finance?.[0]?.card);
    if(!c||!Number.isInteger(c.idx))continue;
    if(c.deck==='chance'&&!held.has('chance:'+c.idx)&&!seenChance.has(c.idx)){
      seenChance.add(c.idx);
      recentChance.push(c.idx);
    }else if(c.deck==='community'&&!held.has('community:'+c.idx)&&!seenCommunity.has(c.idx)){
      seenCommunity.add(c.idx);
      recentCommunity.push(c.idx);
    }
  }

  const bottomChance=new Set(recentChance.slice(0,12));
  const bottomCommunity=new Set(recentCommunity.slice(0,12));

  const chanceRepairSafe=bottomChance.has(11);
  const commRepairSafe=bottomCommunity.has(13);
  const repairSafe=chanceRepairSafe&&commRepairSafe;
  const repairImminent=(!chanceRepairSafe&&recentChance.length>=9)||(!commRepairSafe&&recentCommunity.length>=9);

  const activeMagnetTiles=new Set();
  if(!bottomChance.has(0))activeMagnetTiles.add(39);
  if(!bottomChance.has(2))activeMagnetTiles.add(24);
  if(!bottomChance.has(3))activeMagnetTiles.add(11);
  if(!bottomChance.has(9))activeMagnetTiles.add(19);
  if(!bottomChance.has(13))activeMagnetTiles.add(5);
  if(!bottomChance.has(4)||!bottomChance.has(5)){
    activeMagnetTiles.add(5);
    activeMagnetTiles.add(15);
    activeMagnetTiles.add(25);
  }

  return Object.freeze({
    repairSafe,
    repairImminent,
    activeMagnetTiles,
    recentChanceCount:recentChance.length,
    recentCommunityCount:recentCommunity.length
  });
}

export function situationOf(state,actor,plan){
  const view=observation(state,actor),self=view.players[actor];
  if(!self||self.out)throw Error('NUSA V18.0 actor inactive');
  const rivals=view.players.filter(p=>p.id!==actor&&!p.out);
  const owned=BOARD.filter(t=>t.price&&view.assets[t.id]?.owner===actor);
  const unowned=BOARD.filter(t=>t.price&&view.assets[t.id]?.owner==null);
  const bank=stock(view);
  const cardCycle=analyzePublicCardCycles(view);

  const groups=GROUPS.map((g,i)=>({
    id:i,
    ids:g.ids,
    apexTile:g.ids[g.ids.length-1],
    tier:GROUP_TIER[i]||1,

      id:p.id,
      persona:p.botPersona||'human',
      cash:p.cash,
      count:g.ids.filter(t=>view.assets[t]?.owner===p.id).length
    })),
    complete:g.ids.every(t=>view.assets[t]?.owner===actor&&!view.assets[t]?.mortgaged),
    mortgagedCount:g.ids.filter(t=>view.assets[t]?.owner===actor&&view.assets[t]?.mortgaged).length,
    fullSetOwned:g.ids.every(t=>view.assets[t]?.owner===actor),
    levelMin:Math.min(...g.ids.map(t=>view.assets[t]?.level||0)),
    levelMax:Math.max(...g.ids.map(t=>view.assets[t]?.level||0)),
    redeemCost:g.ids.filter(t=>view.assets[t]?.owner===actor&&view.assets[t]?.mortgaged).reduce((n,t)=>n+redemption(t),0),
    houseCost:BOARD[g.ids[0]]?.build||1000

  // 1-1-1 Split Group Detector for Silent 2-Step Chain Accumulation
  const splitThreeWayGroups=groups
    .filter(g=>g.ids.length===3&&g.unownedCount===0&&g.own===1&&g.owners.filter(o=>o.count===1).length===2)
    .sort((a,b)=>b.tier-a.tier);

  const chainTargetTiles=new Set();
  for(const g of splitThreeWayGroups){
    for(const tid of g.ids){
      if(view.assets[tid]?.owner!==actor&&(view.assets[tid]?.level||0)===0){
        chainTargetTiles.add(tid);
      }

  const collusionMap=new Map();
  for(const g of groups){
    if(g.own!==0||g.unownedCount>0||g.tier<1.12)continue;
    const nearOwner=g.owners.find(o=>o.count===g.ids.length-1);
    const loneOwner=g.owners.find(o=>o.count===1&&o.id!==nearOwner?.id);
    if(nearOwner&&loneOwner){
      const bridgeTile=g.ids.find(tid=>view.assets[tid]?.owner===loneOwner.id&&(view.assets[tid]?.level||0)===0);
      if(bridgeTile!==undefined){
        collusionTargetTiles.add(bridgeTile);
        collusionMap.set(bridgeTile,{
          groupId:g.id,
          tier:g.tier,
          threatRivalId:nearOwner.id,
          holderRivalId:loneOwner.id
        });
      }
    }
  }

  // V18.0 METAGAME #4: Psychological Opponent Memory Profiler (`view.botMemory.opponents`)
  const rawOppMem=view.botMemory?.opponents||{};
  const opponentProfiles=new Map();
  for(const p of rivals){
    const rec=rawOppMem[p.id]||{};
    const obs=rec.observations||0;
    const conf=obs/(obs+6);
    const aggression=1+((rec.aggression||1)-1)*conf;
    const desiredGroups=rec.groups||{};
    const tradeAccepted=rec.tradeAccepted||0;
    const tradeRejected=rec.tradeRejected||0;
    const receptiveness=(tradeAccepted+2)/(tradeAccepted+tradeRejected+4);

      id:p.id,
      aggression,
      confidence:conf,
      desiredGroups,
      receptiveness,
      isHotBidder:aggression>=1.18&&obs>=2
    });
  }

  let visits=Array(40).fill(0),immediateVisits=Array(40).fill(0),ownVisits=Array(40).fill(0);

      visits[i]+=(f2.visits[i]||0)*0.60+(f1.visits[i]||0)*0.40;
    }
  }
  ownVisits=landingForecast(view,actor,1).visits;

  const threat=BOARD.filter(t=>t.price&&view.assets[t.id]?.owner!=null&&view.assets[t.id]?.owner!==actor&&!view.assets[t.id]?.mortgaged);

      return rank(a)-rank(b)||a.price-b.price;
    });
  const liquidBuffer=isolatedMortgageable.reduce((n,t)=>n+(t.mortgage||0),0);

  // Hotel Downgrade Trap Radar (`level === 5 && bank.houses < 4` forces `SELL_SET`)
  const ownHotels=owned.filter(t=>view.assets[t.id]?.level===5).length;

  let forwardHotspotDemand=0;

      const isHotspot=t.type==='station'||(t.group>=0&&HOTSPOT_GROUPS.has(t.group));
      if(isHotspot){
        forwardHotspotDemand+=t.price*(DICE_PROB[d]||0);

      }

  const hotspotReserve=unownedHotspotsAhead>=2?Math.min(2400,Math.round(forwardHotspotDemand*1.85)):0;

  const contestedAnchorTiles=new Set();
  for(const t of unowned){
    if(t.group>=0){
      const g=groups[t.group];
      const rivalHasOne=g.owners.some(o=>o.count===1);
      const rivalAlmostComplete=g.owners.some(o=>o.count===g.ids.length-1);
      if((g.tier>=1.10&&rivalHasOne)||rivalAlmostComplete){
        contestedAnchorTiles.add(t.id);
      }
    }else if(t.type==='station'){
      const rivalHasStation=rivals.some(p=>[5,15,25,35].filter(i=>view.assets[i]?.owner===p.id).length>=1);
      const ownStation=[5,15,25,35].filter(i=>view.assets[i]?.owner===actor).length>=1;
      if(rivalHasStation||ownStation){
        contestedAnchorTiles.add(t.id);
      }
    }
  }

  // Bankruptcy Auction-Queue Lookahead (`view.queue` inspection)
  const queuedAuctionTiles=(view.queue||[])
    .filter(q=>q?.kind==='auction'&&Number.isInteger(q.tile))
    .map(q=>q.tile);

  let queueAuctionReserve=0;
  let hasHighPriorityQueuedAuction=false;
  if(queuedAuctionTiles.length>0){
    const currentAuctionTile=view.phase==='auction'?view.auction?.tile:null;

    for(const qTile of queuedAuctionTiles){
      const qb=BOARD[qTile];
      if(!qb?.price)continue;
      const qg=qb.group>=0?groups[qb.group]:null;
      const qCompletes=qg&&qg.own===qg.ids.length-1;
      const qBlocks=qg&&qg.owners.some(o=>o.count===qg.ids.length-1);

      if(qCompletes||qBlocks||qHighAnchor){

    const f1=rivalForecasts1.get(p.id);
    const riskExposure=owned.filter(t=>!view.assets[t.id]?.mortgaged).reduce((acc,t)=>acc+((f1?.visits[t.id]||0)*rent(view,t.id,7)),0);
    return {id:p.id,cash:p.cash,liquid:pLiquid,deedCount:pDeeds.length,riskExposure,vulnerable:p.cash<1800||pLiquid<3500};

    hasPendingRivalRentClaim||((view.doubles||0)===0&&bank.houses>4&&!hotelTrapActive)

          earlyCashSinkTiles.add(t.id);
        }
      }else if(t.type==='utility'){
        const ownUtils=[12,28].filter(i=>view.assets[i]?.owner===actor).length;
        if(ownUtils===0&&!steppingStoneTiles.has(t.id)&&(self.cash-t.price<trueReserve+hotspotReserve+1400)){
          earlyCashSinkTiles.add(t.id);
        }
      }
    }
  }

  const shares=strengthShares(view,BOARD,GROUPS,rent);
  const leader=rivals.slice().sort((a,b)=>(shares[b.id]-shares[a.id])||(b.cash-a.cash))[0]||null;

  const ownMonopolies=groups.filter(g=>g.fullSetOwned).length;
  const rivalMonopolies=rivals.filter(p=>GROUPS.some(g=>g.ids.every(id=>view.assets[id]?.owner===p.id))).length;
  const boardDeadlocked=unowned.length<=6&&ownMonopolies===0&&rivalMonopolies<=1;

  const hasMortgagedMonopoly=groups.some(g=>g.fullSetOwned&&g.mortgagedCount>0);
  const hasUnbuiltMonopoly=groups.some(g=>g.complete&&g.levelMin<3);
  const needsBuildCapital=hasMortgagedMonopoly||hasUnbuiltMonopoly;
  const mortgagedOwned=owned.filter(t=>view.assets[t.id]?.mortgaged);
  const canDeleverage=!needsBuildCapital&&mortgagedOwned.length>0&&self.cash>trueReserve+1200;

  // Same-Turn Multi-Trade Chain Counter
  const roundTradeCount=(view.log||[]).filter(e=>e.type==='trade'&&e.turn===view.round&&e.actor===actor).length;

  // Pre-Buy Flash Liquidity & V18.0 Cross-Phase Buy-Rescue Trade Detector (when in phase === 'buy')
  let needsPreBuyMortgage=false;
  let needsPreBuyTrade=false;
  if(view.phase==='buy'&&view.turn===actor&&Number.isInteger(view.buy)){
    const bTile=BOARD[view.buy];
    if(bTile?.price&&!earlyCashSinkTiles.has(view.buy)&&self.cash<bTile.price){
      if(self.cash+liquidBuffer>=bTile.price){
        needsPreBuyMortgage=true;
      }else if(canTrade(view,actor)&&roundTradeCount<2&&isolatedMortgageable.length>0){
        needsPreBuyTrade=true;

export function completion(c,t,who=c.actor){
  const b=BOARD[t];
  if(!b)return 0;
  if(b.type==='station'){
    const ownCount=[5,15,25,35].filter(i=>owner(c,i)===who&&i!==t).length;
    return clamp(0.32+ownCount*0.24,0,1);
  }
  if(b.type==='utility'){
    const ownCount=[12,28].filter(i=>owner(c,i)===who&&i!==t).length;
    return ownCount>=1?0.40:0.10;
  }
  if(b.group<0)return 0;
  const ids=GROUPS[b.group].ids;
  const ownCount=ids.filter(i=>owner(c,i)===who&&i!==t).length;
  const otherCount=ids.filter(i=>owner(c,i)!=null&&owner(c,i)!==who&&i!==t).length;
  if(ownCount===ids.length-1)return 1.0;
  if(otherCount===0&&ownCount>0)return (ownCount/ids.length)*0.84;
  if(otherCount===0)return 0.36;
  return (ownCount/ids.length)*0.28;
}

export function block(c,t){
  const b=BOARD[t];
  if(!b)return 0;
  if(b.type==='station'){
    const rivalMax=Math.max(0,...c.rivals.map(p=>[5,15,25,35].filter(i=>owner(c,i)===p.id&&i!==t).length));
    return rivalMax>=2?0.72:rivalMax===1?0.35:0;
  }
  if(b.group<0)return 0;
  const g=c.groups[b.group];
  const tier=GROUP_TIER[b.group]||1;
  let maxThreat=0;
  for(const o of g.owners){
    const isPriorityRival=(o.persona==='bima'||o.persona==='kresna')||(c.leader?.id===o.id);
    if(o.count===g.ids.length-1){
      const rivalCash=c.view.players[o.id]?.cash||0;
      const readiness=clamp(rivalCash/Math.max(1000,g.ids.length*2*g.houseCost),0.35,1.25);
      const boost=isPriorityRival?1.30:1.0;
      maxThreat=Math.max(maxThreat,clamp(0.95*readiness*boost,0,1));
    }else if(o.count===1&&g.own===0){
      const anchorVal=tier>=1.25?(isPriorityRival?0.78:0.68):tier>=1.10?0.52:0.28;
      maxThreat=Math.max(maxThreat,anchorVal);
    }
  }
  return maxThreat;
}

export function valuation(c,t){

  if(!p)return 0;
  if(p.type==='utility'){
    const ownUt=[12,28].filter(i=>owner(c,i)===c.actor&&i!==t).length;
    return p.price*(ownUt>=1?0.96:0.74);
  }
  if(p.type==='station'){
    const ownSt=[5,15,25,35].filter(i=>owner(c,i)===c.actor&&i!==t).length;
    const geoMult=STATION_GEOMETRY_MULT[t]||1.0;
    const fifoBoost=c.cardCycle?.activeMagnetTiles?.has(t)?p.price*0.14:0;
    return p.price*(1.05+ownSt*0.36)*geoMult+fifoBoost;
  }
  const tier=p.group>=0?(GROUP_TIER[p.group]||1):1;
  const synergy=completion(c,t,c.actor);
  const denial=block(c,t);
  const apexBonus=APEX_TILES.has(t)?p.price*0.08*tier:0;
  const anchorBonus=c.contestedAnchorTiles?.has(t)?p.price*0.28*tier:0;
  const stepBonus=c.steppingStoneTiles?.has(t)?p.price*0.32:0;

  const collusionBonus=c.collusionTargetTiles?.has(t)?p.price*0.34*tier:0;

export function grandmasterAuctionCeiling(c){

  if(auction.kind==='building'){
    const t=auction.targets?.[c.actor];
    if(!Number.isInteger(t))return 0;
    const bCost=BOARD[t]?.build||1000;
    const lv=c.view.assets[t]?.level||0;
    const mult=lv===2?2.25:lv===1?2.05:lv===3?1.85:1.55;
    return Math.min(Math.max(0,c.self.cash-c.trueReserve*0.35),Math.floor((bCost*mult)/10)*10);
  }
  const t=auction.tile,p=BOARD[t];
  if(!p)return 0;
  const comp=completion(c,t,c.actor);
  const blk=block(c,t);
  const tier=p.group>=0?(GROUP_TIER[p.group]||1):1;

  let maxVal=p.price*0.98;
  if(comp>=0.95){
    maxVal=p.price*Math.min(1.55,Math.max(1.30,tier*1.12))+350;
  }else if(blk>=0.80){
    const needyRival=c.groups[p.group]?.owners?.find(o=>o.count===GROUPS[p.group].ids.length-1);
    const rivalStillBidding=needyRival&&auction.participants?.includes(needyRival.id)&&!auction.passed?.includes(needyRival.id);
    const rivalCash=needyRival?c.view.players[needyRival.id]?.cash||0:0;
    const prof=needyRival?c.opponentProfiles?.get(needyRival.id):null;
    const driveCap=prof?.isHotBidder?p.price*1.36+220:p.price*1.32+200;
    maxVal=rivalStillBidding?Math.min(rivalCash+10,driveCap):p.price*1.08;
  }else if(c.earlyCashSinkTiles?.has(t)){
    maxVal=p.price*0.82;
  }else if(c.contestedAnchorTiles?.has(t)||c.collusionTargetTiles?.has(t)){
    maxVal=p.price*Math.min(1.08,Math.max(0.98,tier*0.84));
  }else if(c.steppingStoneTiles?.has(t)||c.chainTargetTiles?.has(t)){
    maxVal=p.price*1.08;
  }else if(comp>=0.40){
    maxVal=p.price*Math.min(1.06,Math.max(0.98,tier*0.88));
  }else if(p.type==='station'){
    const ownSt=[5,15,25,35].filter(i=>owner(c,i)===c.actor&&i!==t).length;
    const geoMult=STATION_GEOMETRY_MULT[t]||1.0;
    maxVal=p.price*(0.96+ownSt*0.10)*geoMult;

  const safeCash=Math.max(0,c.self.cash+(c.liquidBuffer||0)*0.35-c.trueReserve*0.45-qReserve);

export function normalizeTrade(c,a){
  if(!['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(a.type))return null;
  const raw=(a.type==='TRADE_ACCEPT'||a.type==='TRADE_REJECT')?c.view.trade:a.trade;
  if(!raw)return null;
  const isIncoming=a.type==='TRADE_ACCEPT'||a.type==='TRADE_REJECT';
  const proposerId=isIncoming?raw.from:c.actor;
  const partnerId=isIncoming?raw.from:raw.to;
  const myGive=isIncoming?(raw.take||[]):(raw.give||[]);
  const myTake=isIncoming?(raw.give||[]):(raw.take||[]);
  const cashReceived=isIncoming?(raw.giveCash||0):(raw.takeCash||0);
  const cashPaid=isIncoming?(raw.takeCash||0):(raw.giveCash||0);
  const cardsReceived=isIncoming?(raw.giveCards||0):(raw.takeCards||0);
  const cardsPaid=isIncoming?(raw.takeCards||0):(raw.giveCards||0);
  const mortgageFeePaid=myTake.reduce((n,t)=>n+(c.view.assets[t]?.mortgaged?interest(t):0),0);
  const mortgageFeePartner=myGive.reduce((n,t)=>n+(c.view.assets[t]?.mortgaged?interest(t):0),0);
  return {
    raw,isIncoming,isReject:a.type==='TRADE_REJECT',
    proposerId,partnerId,myGive,myTake,
    netCash:cashReceived-cashPaid-mortgageFeePaid,
    partnerNetCash:cashPaid-cashReceived-mortgageFeePartner,
    netCards:cardsReceived-cardsPaid
  };
}

export function completesGroupPostTrade(c,playerId,acquired,surrendered){
  const acqSet=new Set(acquired),surSet=new Set(surrendered);
  let count=0,bestTier=0,buildCost3=0,groupIds=[];
  for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
    const ids=GROUPS[gIdx].ids;
    const hadBefore=ids.every(i=>owner(c,i)===playerId);
    const hasAfter=ids.every(i=>(owner(c,i)===playerId&&!surSet.has(i))||acqSet.has(i));
    if(hasAfter&&!hadBefore){
      count++;
      groupIds.push(gIdx);
      bestTier=Math.max(bestTier,GROUP_TIER[gIdx]||1);
      buildCost3+=ids.length*3*(BOARD[ids[0]]?.build||1000);
    }
  }
  return {count,bestTier,buildCost3,groupIds};
}

export function getBilateralTradeEvaluation(c,trade,proposerId=c.actor){
  const partnerId=proposerId===c.actor?trade.to:proposerId;
  const key=proposerId+':'+partnerId+':'+JSON.stringify(trade);
  if(c._tradeEvalCache?.has(key))return c._tradeEvalCache.get(key);

  let partnerAnalysis=c._partnerAnalyses?.get(partnerId);
  if(!partnerAnalysis){
    partnerAnalysis=analyzeState(c.view,partnerId,{fast:false});
    c._partnerAnalyses?.set(partnerId,partnerAnalysis);
  }
  const proposerPublic=analyzeState(c.view,proposerId,{fast:true,publicOpponent:true});
  const partnerContext={
    observerId:partnerId,
    analyses:new Map([[partnerId,partnerAnalysis],[proposerId,proposerPublic]])
  };

  const myAnalysis=c.ownAnalysis||analyzeState(c.view,c.actor,{fast:false});
  const partnerPublic=analyzeState(c.view,partnerId,{fast:true,publicOpponent:true});
  const myContext={
    observerId:c.actor,
    analyses:new Map([[c.actor,myAnalysis],[partnerId,partnerPublic]])
  };

  const partnerEval=evaluateTrade(c.view,partnerId,trade,proposerId,partnerContext);
  const myEval=evaluateTrade(c.view,c.actor,trade,proposerId,myContext);
  const result={myEval,partnerEval};
  c._tradeEvalCache?.set(key,result);
  return result;
}

export function predictPartnerAcceptance(c,trade){
  const partnerId=trade.to;
  const partner=c.view.players[partnerId];
  if(!partner||partner.out)return {willAccept:false,partnerNet:-1e8,myEval:null,partnerEval:null};

  const {myEval,partnerEval}=getBilateralTradeEvaluation(c,trade,c.actor);
  if(partnerEval.invalid||partnerEval.insolventAfterFees||partnerEval.unilateralLoss||partnerEval.net<2){
    return {willAccept:false,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  const myMono=completesGroupPostTrade(c,c.actor,trade.take||[],trade.give||[]);
  const oppMono=completesGroupPostTrade(c,partnerId,trade.give||[],trade.take||[]);
  const persona=partner.botPersona||'human';

  if(persona==='kresna'){
    if(myMono.count>0&&oppMono.count===0){
      return {willAccept:false,partnerNet:partnerEval.net,myEval,partnerEval};
    }
    return {willAccept:partnerEval.net>=15,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  if(persona==='apex'){
    return {willAccept:partnerEval.net>=5,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  const requiredNet=myMono.count>0&&oppMono.count===0?75:12;
  return {
    willAccept:partnerEval.net>=requiredNet,
    partnerNet:partnerEval.net,
    myEval,
    partnerEval
  };
}

export function evaluateTradeForActor(c,norm){
  if(!norm)return 0;
  const myPropGain=norm.myTake.reduce((n,t)=>n+valuation(c,t),0)-norm.myGive.reduce((n,t)=>n+valuation(c,t),0);
  const faceDelta=norm.myTake.reduce((n,t)=>n+BOARD[t].price,0)-norm.myGive.reduce((n,t)=>n+BOARD[t].price,0)+norm.netCash;
  // Jail card value adapts to stage: worth 450 in early game, 180 in mid/late game when staying in jail is desirable!
  const cardUnitVal=c.canGiveJailCardSweetener?180:420;
  const cashAndCardGain=norm.netCash+norm.netCards*cardUnitVal;
  const myMono=completesGroupPostTrade(c,c.actor,norm.myTake,norm.myGive);
  const oppMono=completesGroupPostTrade(c,norm.partnerId,norm.myGive,norm.myTake);

  const myPostCash=c.self.cash+norm.netCash+(c.liquidBuffer||0)*0.75;
  const oppPlayer=c.view.players[norm.partnerId];
  const oppPostCash=(oppPlayer?.cash||0)+norm.partnerNetCash;

  let strategicDelta=0;
  if(oppMono.count>0&&myMono.count===0){
    strategicDelta-=6000;
  }else if(myMono.count>0&&oppMono.count===0){
    const houseCost=BOARD[GROUPS[myMono.groupIds[0]].ids[0]]?.build||1000;
    if(myPostCash<houseCost*2||faceDelta<-950){
      strategicDelta-=2800;
    }else{
      const buildReadiness=clamp(myPostCash/Math.max(1500,myMono.buildCost3*0.55),0.45,1.40);
      strategicDelta+=3500*myMono.bestTier*buildReadiness;
    }
  }else if(myMono.count>0&&oppMono.count>0){
    const myHouseCost=BOARD[GROUPS[myMono.groupIds[0]].ids[0]]?.build||1000;
    const oppHouseCost=BOARD[GROUPS[oppMono.groupIds[0]].ids[0]]?.build||1000;
    const myBuildableHouses=Math.min(9,Math.floor(Math.max(0,myPostCash-c.trueReserve*0.4)/myHouseCost));
    const oppBuildableHouses=Math.min(9,Math.floor(Math.max(0,oppPostCash-300)/oppHouseCost));
    const firstBuildAdvantage=!norm.isIncoming&&myBuildableHouses>=3;

    if(firstBuildAdvantage&&(myBuildableHouses>=oppBuildableHouses+2||oppBuildableHouses<=1||myMono.bestTier>oppMono.bestTier*1.12)){
      strategicDelta+=3200;
    }else if(norm.isIncoming&&oppBuildableHouses>=2){
      strategicDelta-=4200;
    }else{
      strategicDelta-=3600;
    }
  }else{
    if((c.view.phase==='debt'||(c.view.phase==='buy'&&c.needsPreBuyTrade))&&norm.netCash>0&&faceDelta>=-320){
      strategicDelta+=2400;
    }else if(faceDelta<-250){
      strategicDelta-=1800;
    }else{
      for(const t of norm.myTake){
        if(c.collusionTargetTiles?.has(t))strategicDelta+=1050*(GROUP_TIER[BOARD[t]?.group]||1);
        if(c.steppingStoneTiles?.has(t))strategicDelta+=950;
        if(c.chainTargetTiles?.has(t))strategicDelta+=820*(GROUP_TIER[BOARD[t]?.group]||1);
        const b=BOARD[t];
        if(b?.group>=0&&c.groups[b.group].own===1&&GROUPS[b.group].ids.length===3){
          strategicDelta+=480*(GROUP_TIER[b.group]||1);
        }
      }
      if(c.needsBuildCapital&&norm.netCash>=900&&faceDelta>=-50&&norm.myGive.every(t=>!c.groups[BOARD[t]?.group]?.fullSetOwned)){
        strategicDelta+=750;
      }
    }
  }

  const rawScore=(myPropGain+cashAndCardGain+strategicDelta)/Math.max(1200,c.salary);
  const signal=clamp(rawScore,-1,1);

function sovereignScoreRebase(context,action,nativeScore){
  const {view,self,trueReserve,liquidBuffer}=context;
  const inflatedReserve=Math.max(trueReserve*1.6,self.cash*0.55);
  let spend=0;
  if(action.type==='BUILD'||action.type==='BUILD_JOIN'){
    if(context.shouldDeferPreRollBuild&&action.type==='BUILD')return nativeScore-450;
    spend=BOARD[action.tile]?.build||0;
  }else if(action.type==='BUY'){
    if(context.earlyCashSinkTiles?.has(view.buy))return nativeScore;
    spend=BOARD[view.buy]?.price||0;
  }else if(action.type==='REDEEM'||action.type==='MORTGAGE_REDEEM'){
    if(context.shouldDeferPreRollBuild&&action.type==='REDEEM')return nativeScore-380;
    const t=action.type==='MORTGAGE_REDEEM'?view.mortgageDecision?.tile:action.tile;
    const g=BOARD[t]?.group;
    if(Number.isInteger(t)&&((g>=0&&context.groups[g]?.fullSetOwned)||context.canDeleverage||action.type==='MORTGAGE_REDEEM')){
      spend=redemption(t);
    }
  }
  if(spend<=0)return nativeScore;
  const effectiveAfter=self.cash-spend+liquidBuffer*0.75;
  if(effectiveAfter<trueReserve)return nativeScore;

  const penaltyBefore=liquidityPenalty(self.cash,inflatedReserve);
  const penaltyAfter=liquidityPenalty(self.cash-spend,inflatedReserve);
  const artificialDrag=Math.max(0,penaltyAfter-penaltyBefore);
  return nativeScore+artificialDrag*0.85;
}

        return desB-desA||a.price-b.price;
      });

      let hijackSynthesized=false;
      for(const targetDeed of senderTargets.slice(0,2)){
        if(hijackSynthesized)break;
        const giveOptions=safeBaitsForSender.slice(0,2).map(b=>[b.id]);
        giveOptions.push([]);
        for(const giveList of giveOptions){
          if(hijackSynthesized)break;
          const giveVal=giveList.reduce((n,id)=>n+BOARD[id].price,0);
          const cardSweetener=(context.canGiveJailCardSweetener&&(self.cards?.length||0)>0)?1:0;
          for(const overpay of [0,150,300,450]){
            const rawPay=targetDeed.price-giveVal+overpay-(cardSweetener*250);
            const giveCash=rawPay>0?Math.round(rawPay/10)*10:0;
            if(giveList.length===0&&giveCash<targetDeed.price&&cardSweetener===0)continue;
            if(self.cash-giveCash<context.trueReserve+1200)continue;
            const counterTrade={
              to:senderId,
              give:giveList,
              take:[targetDeed.id],
              giveCash,
              takeCash:0,
              giveCards:cardSweetener,
              takeCards:0
            };
            const {myEval,partnerEval}=getBilateralTradeEvaluation(context,counterTrade,actor);
            if(myEval.net>0&&partnerEval.net>=15&&!partnerEval.invalid&&!partnerEval.insolventAfterFees){
              if(pushCandidate({type:'COUNTER',trade:counterTrade,actionId:'nusa-counter-hijack-'+targetDeed.id},topScore()+410)){
                hijackSynthesized=true;
                break;
              }
            }
          }
        }
      }

      // Standard Cash Counter if incoming trade is weak and no hijack was synthesized
      if(!hijackSynthesized&&evalScore<0.08){
        const askExtra=Math.min(senderCash,Math.max(300,Math.round((t.takeCash||0)+450)));
        if(askExtra>0&&(t.take?.length||0)>0){
          const counterTrade={
            to:senderId,
            give:[...(t.take||[])],
            take:[...(t.give||[])],
            giveCash:0,
            takeCash:askExtra,
            giveCards:0,
            takeCards:0
          };
          const {myEval,partnerEval}=getBilateralTradeEvaluation(context,counterTrade,actor);
          if(myEval.net>0&&partnerEval.net>=10){
            pushCandidate({type:'COUNTER',trade:counterTrade,actionId:'nusa-counter-cash'},topScore()+180);
          }
        }
      }
    }
  }

  // 2B. DEBT-PHASE EMERGENCY TRADE RESCUE (Mode E: Save Monopoly Houses from 50% Liquidation Loss)
  if(phase==='debt'&&view.debt?.from===actor&&self.cash<view.debt.amount&&canTrade(view,actor)&&context.roundTradeCount<2){
    const deficit=view.debt.amount-self.cash;
    if(context.liquidBuffer<deficit&&context.isolatedMortgageable.length>0){
      const mem=memoryFor(view,actor);
      const richRivals=context.rivals.filter(p=>p.cash>=400&&!rejectionActive(view,mem,p.id)).sort((a,b)=>b.cash-a.cash);
      let debtTradeSynthesized=false;
      for(const r of richRivals){
        if(debtTradeSynthesized)break;
        for(const deed of context.isolatedMortgageable){
          if(deed.group>=0){
            const oppCnt=GROUPS[deed.group].ids.filter(i=>view.assets[i]?.owner===r.id).length;
            if(oppCnt>=GROUPS[deed.group].ids.length-1)continue;
          }
          for(const mult of [1.0,0.92,0.85]){
            const askCash=Math.min(r.cash,Math.max(deed.mortgage+100,Math.floor((deed.price*mult)/10)*10));
            if(askCash<=deed.mortgage)continue;
            const trade={to:r.id,give:[deed.id],take:[],giveCash:0,takeCash:askCash,giveCards:0,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            const oracle=predictPartnerAcceptance(context,trade);
            if(oracle.willAccept){
              pushCandidate({type:'OFFER',trade,actionId:'nusa-debt-rescue-'+deed.id},topScore()+480);
              debtTradeSynthesized=true;
              break;
            }
          }
          if(debtTradeSynthesized)break;
        }
      }
    }
  }

  // 3. EARLY JAIL BLITZ & POST-ROLL DISCIPLINED ASSET MANAGEMENT ('roll', 'end', 'jail')
  if(phase==='jail'&&view.turn===actor&&context.unowned.length>=5&&context.nearLoss<420){
    if(self.cards?.length>0){
      pushCandidate({type:'JAIL_CARD',actionId:'nusa-early-jail-card'},topScore()+260);
    }else if(self.cash-500>=context.trueReserve+1200){
      pushCandidate({type:'JAIL_PAY',actionId:'nusa-early-jail-pay'},topScore()+240);
    }
  }

  if(phase==='roll'&&view.turn===actor&&context.shouldDeferPreRollBuild){
    pushCandidate({type:'ROLL',actionId:'nusa-disciplined-roll'},topScore()+(context.hasPendingRivalRentClaim?480:350));
  }

  if(['roll','end','jail'].includes(phase)&&view.turn===actor&&canManage(view,actor)&&!context.shouldDeferPreRollBuild){
    // 3A. Unmortgage any mortgaged tile inside a completed color set!
    for(const g of context.groups){
      if(g.fullSetOwned&&g.mortgagedCount>0){
        const mTile=g.ids.find(id=>view.assets[id]?.mortgaged);
        if(mTile!==undefined){
          const rCost=redemption(mTile);
          if(self.cash-rCost>=context.trueReserve*0.45&&!managementError(view,actor,mTile,'REDEEM')){
            pushCandidate({type:'REDEEM',tile:mTile,actionId:'nusa-redeem-mono-'+mTile},topScore()+400);
          }
        }
      }
    }

    // 3B. Aggressive 3-House Construction, Hotel-Trap Lock, & Apex-Tile Even-Build Ordering!
    const maxTargetLevel=4;
    const buildGroups=context.groups
      .filter(g=>g.complete&&g.levelMin<maxTargetLevel)
      .sort((a,b)=>(GROUP_TIER[b.id]||1)-(GROUP_TIER[a.id]||1));

    for(const g of buildGroups){
      const levelMin=g.levelMin;
      const lockHouses=(context.sariPresent||context.rivalMonopolies>0||context.rivalHasHotel)?context.bank.houses<=18:context.bank.houses<=14;
      if(levelMin===4&&lockHouses)continue;

      const validTiles=g.ids
        .filter(id=>!managementError(view,actor,id,'BUILD'))
        .sort((x,y)=>{
          const apexX=APEX_TILES.has(x)?0.45:0;
          const apexY=APEX_TILES.has(y)?0.45:0;
          const scoreX=(context.tileKillshot[x]||0)*2.5+(context.immediateVisits[x]||0)+apexX;
          const scoreY=(context.tileKillshot[y]||0)*2.5+(context.immediateVisits[y]||0)+apexY;
          return scoreY-scoreX;
        });

      if(validTiles.length>0){
        const bestTile=validTiles[0];
        const bCost=BOARD[bestTile].build;
        const safetyFloor=(levelMin<3||context.hotelTrapActive)?context.trueReserve*0.48:context.trueReserve*0.80;
        if(self.cash-bCost>=safetyFloor){
          const trapBoost=context.hotelTrapActive?415:(levelMin<3?395:240);
          pushCandidate({type:'BUILD',tile:bestTile,actionId:'nusa-build-'+bestTile},topScore()+trapBoost);
        }else if(levelMin<3||context.hotelTrapActive){
          const utilCollateral=context.isolatedMortgageable.find(t=>t.type==='utility');
          if(utilCollateral&&self.cash+utilCollateral.mortgage-bCost>=safetyFloor&&!managementError(view,actor,utilCollateral.id,'MORTGAGE')){
            pushCandidate({type:'MORTGAGE',tile:utilCollateral.id,actionId:'nusa-leverage-util-'+utilCollateral.id},topScore()+250);
          }
        }
      }
    }

    // 3C. Late-Game De-Leveraging: Redeem Stations & Mortgaged Deeds when Flush with Cash
    if(context.canDeleverage&&phase==='end'){
      const redeemTargets=context.mortgagedOwned
        .filter(t=>!managementError(view,actor,t.id,'REDEEM')&&self.cash-redemption(t.id)>=context.trueReserve+900)
        .sort((a,b)=>(b.type==='station'?1:0)-(a.type==='station'?1:0)||b.price-a.price);
      if(redeemTargets.length>0){
        const rt=redeemTargets[0];
        pushCandidate({type:'REDEEM',tile:rt.id,actionId:'nusa-deleverage-'+rt.id},topScore()+220);
      }
    }
  }

  // 4. OMNISCIENT MIND-READER TRADE SYNTHESIZER (V18.0 Multi-Trade Chaining + Collusion Interdiction + Jail-Card Arbitrage!)
  if(phase==='end'&&view.turn===actor&&canTrade(view,actor)&&context.roundTradeCount<3){
    for(const c of candidates){
      if(c.action.type==='OFFER'){
        const norm=normalizeTrade(context,c.action);
        const s=evaluateTradeForActor(context,norm);
        const oracle=predictPartnerAcceptance(context,c.action.trade);
        if(oracle.willAccept&&s>=0.22&&oracle.myEval.net>0&&oracle.myEval.deltaShare>=oracle.myEval.partnerDelta){
          c.score=Math.max(c.score+220,topScore()+200*s);
        }else if(!oracle.willAccept){
          c.score-=1500;
        }
      }
    }

    const mem=memoryFor(view,actor);
    const groupOrder=[3,4,1,2,5,7,0,6];
    let synthesizedCount=0;

    // Mode A & B: 2/3 -> 3/3 Monopoly Buyout (with Psychological Bait Sorting & Jail-Card Arbitrage!) & Asymmetric Swap
    for(const gId of groupOrder){
      if(synthesizedCount>=3)break;
      const g=context.groups[gId];
      if(g.own!==g.ids.length-1||g.unownedCount>0)continue;
      const missingTile=g.ids.find(id=>view.assets[id]?.owner!==actor);
      if(missingTile===undefined)continue;
      const targetOwner=view.assets[missingTile]?.owner;
      if(targetOwner==null||targetOwner===actor||rejectionActive(view,mem,targetOwner))continue;
      if(g.ids.some(id=>(view.assets[id]?.level||0)>0))continue;

      const tPrice=BOARD[missingTile].price;
      const minBuildReserve=g.houseCost*2+context.trueReserve*0.50;
      const maxCashOffer=Math.max(0,Math.floor((self.cash-minBuildReserve)/10)*10);
      const partnerMaxCash=view.players[targetOwner]?.cash||0;
      const partnerDesire=context.opponentProfiles?.get(targetOwner)?.desiredGroups||{};

      const safeBaits=context.owned.filter(d=>{
        if(d.group===gId||(view.assets[d.id]?.level||0)>0)return false;
        if(d.group>=0&&GROUPS[d.group].ids.some(i=>(view.assets[i]?.level||0)>0))return false;
        if(d.group>=0){
          const oppCount=GROUPS[d.group].ids.filter(i=>view.assets[i]?.owner===targetOwner).length;
          if(oppCount>=GROUPS[d.group].ids.length-1)return false;
        }
        return true;
      }).sort((a,b)=>{
        const desA=a.group>=0?(partnerDesire[a.group]||0):0;
        const desB=b.group>=0?(partnerDesire[b.group]||0):0;
        if(desB!==desA)return desB-desA;
        const oppHasA=a.group>=0?GROUPS[a.group].ids.filter(i=>view.assets[i]?.owner===targetOwner).length:0;
        const oppHasB=b.group>=0?GROUPS[b.group].ids.filter(i=>view.assets[i]?.owner===targetOwner).length:0;
        return oppHasB-oppHasA||a.price-b.price;
      });

      const packages=[];
      for(const b of safeBaits.slice(0,3))packages.push([b.id]);
      if(safeBaits.length>=2)packages.push([safeBaits[0].id,safeBaits[1].id]);
      packages.push([]);

      const cardOptions=(context.canGiveJailCardSweetener&&(self.cards?.length||0)>0)?[1,0]:[0];
      let foundForGroup=false;
      for(const giveList of packages){
        if(foundForGroup)break;
        const giveVal=giveList.reduce((n,id)=>n+BOARD[id].price,0);
        for(const giveCards of cardOptions){
          if(foundForGroup)break;
          const cardVal=giveCards*250;
          for(const netOverpay of [-300,-150,0,100,200,320,450,600]){
            const rawPayment=tPrice-giveVal-cardVal+netOverpay;
            const giveCash=rawPayment>0?Math.min(maxCashOffer,Math.round(rawPayment/10)*10):0;
            const takeCash=rawPayment<0?Math.min(partnerMaxCash,Math.round(Math.abs(rawPayment)/10)*10):0;
            if(giveList.length===0&&giveCards===0&&giveCash<tPrice)continue;
            if(rawPayment>maxCashOffer)continue;

            const trade={to:targetOwner,give:giveList,take:[missingTile],giveCash,takeCash,giveCards,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            const oracle=predictPartnerAcceptance(context,trade);
            if(oracle.willAccept&&oracle.myEval.net>0&&oracle.myEval.deltaShare>=oracle.myEval.partnerDelta){
              const action={type:'OFFER',trade,actionId:'nusa-oracle-buyout-'+missingTile+'-'+giveCash+'-c'+giveCards};
              if(pushCandidate(action,topScore()+390+(giveCards*15))){
                synthesizedCount++;
                foundForGroup=true;
                break;
              }
            }
          }
        }
      }

      if(!foundForGroup&&synthesizedCount<3){
        const swapBaits=context.owned.filter(d=>{
          if(d.group===gId||d.group<0||(view.assets[d.id]?.level||0)>0)return false;
          if(GROUPS[d.group].ids.some(i=>(view.assets[i]?.level||0)>0))return false;
          const oppCount=GROUPS[d.group].ids.filter(i=>view.assets[i]?.owner===targetOwner).length;
          return oppCount===GROUPS[d.group].ids.length-1;
        });
        for(const sb of swapBaits){
          const oppHouseCost=BOARD[sb.id]?.build||1000;
          const askCash=Math.min(partnerMaxCash,Math.max(0,Math.floor(((view.players[targetOwner]?.cash||0)-oppHouseCost)/10)*10));
          for(const takeCash of [askCash,Math.floor(askCash*0.6/10)*10,0]){
            const oppPostCash=(view.players[targetOwner]?.cash||0)-takeCash;
            const myPostCash=self.cash+takeCash;
            const myHouses=Math.floor(Math.max(0,myPostCash-context.trueReserve*0.4)/g.houseCost);
            const oppHouses=Math.floor(Math.max(0,oppPostCash-300)/oppHouseCost);
            if(myHouses<3||(myHouses<oppHouses+2&&oppHouses>1))continue;

            const trade={to:targetOwner,give:[sb.id],take:[missingTile],giveCash:0,takeCash,giveCards:0,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            const oracle=predictPartnerAcceptance(context,trade);
            if(oracle.willAccept&&oracle.myEval.net>0&&oracle.myEval.deltaShare>=oracle.myEval.partnerDelta){
              const action={type:'OFFER',trade,actionId:'nusa-asym-swap-'+missingTile+'-'+sb.id};
              if(pushCandidate(action,topScore()+340)){
                synthesizedCount++;
                foundForGroup=true;
                break;
              }
            }
          }
          if(foundForGroup)break;
        }
      }
    }

    // Mode D: Silent 2-Step Chain Accumulation (1/3 -> 2/3 when group is split 1-1-1!)
    if(synthesizedCount===0&&context.splitThreeWayGroups.length>0){
      for(const sg of context.splitThreeWayGroups){
        if(synthesizedCount>=2)break;
        const rivalTiles=sg.ids.filter(id=>view.assets[id]?.owner!==actor&&(view.assets[id]?.level||0)===0);
        for(const rTile of rivalTiles){
          if(synthesizedCount>=2)break;
          const rOwner=view.assets[rTile]?.owner;
          if(rOwner==null||rejectionActive(view,mem,rOwner))continue;
          const rPrice=BOARD[rTile].price;

          const chainBaits=context.isolatedMortgageable.filter(d=>{
            if(d.group===sg.id)return false;
            if(d.group>=0&&context.groups[d.group].own>=2)return false;
            if(d.group>=0&&(GROUP_TIER[d.group]||1)>sg.tier)return false;
            if(d.group>=0){
              const oppCnt=GROUPS[d.group].ids.filter(i=>view.assets[i]?.owner===rOwner).length;
              if(oppCnt>=GROUPS[d.group].ids.length-1)return false;
            }
            return true;
          }).sort((a,b)=>{
            const desA=context.opponentProfiles?.get(rOwner)?.desiredGroups?.[a.group]||0;
            const desB=context.opponentProfiles?.get(rOwner)?.desiredGroups?.[b.group]||0;
            if(desB!==desA)return desB-desA;
            const oppHasA=a.group>=0?GROUPS[a.group].ids.filter(i=>view.assets[i]?.owner===rOwner).length:0;
            const oppHasB=b.group>=0?GROUPS[b.group].ids.filter(i=>view.assets[i]?.owner===rOwner).length:0;
            return oppHasB-oppHasA||a.price-b.price;
          });

          let chainFound=false;
          for(const cb of chainBaits.slice(0,3)){
            if(chainFound)break;
            for(const sweet of [-100,0,60,120,180]){
              const rawDiff=rPrice-cb.price+sweet;
              const giveCash=rawDiff>0?Math.round(rawDiff/10)*10:0;
              const takeCash=rawDiff<0?Math.min(view.players[rOwner]?.cash||0,Math.round(Math.abs(rawDiff)/10)*10):0;
              if(self.cash-giveCash<context.trueReserve+1200)continue;
              const trade={to:rOwner,give:[cb.id],take:[rTile],giveCash,takeCash,giveCards:0,takeCards:0};
              if(JSON.stringify(trade)===mem.lastSignature)continue;
              const oracle=predictPartnerAcceptance(context,trade);
              if(oracle.willAccept&&oracle.myEval.net>=0){
                const action={type:'OFFER',trade,actionId:'nusa-chain-accum-'+rTile};
                if(pushCandidate(action,topScore()+315)){
                  synthesizedCount++;
                  chainFound=true;
                  break;
                }
              }
            }
          }
        }
      }
    }

    // Mode F (V18.0 METAGAME #2): Preemptive Rival-Collusion Interdiction!
    // If Rival A has 2/3 of a dangerous color group and Rival B holds the 3rd isolated tile,
    // NUSA buys that 3rd tile from Rival B before Rival A can trade for it!
    if(synthesizedCount===0&&context.collusionTargetTiles.size>0){
      for(const bridgeTile of context.collusionTargetTiles){
        if(synthesizedCount>=2)break;
        const info=context.collusionMap.get(bridgeTile);
        const holderId=view.assets[bridgeTile]?.owner;
        if(!info||holderId==null||rejectionActive(view,mem,holderId))continue;
        if(info.tier<1.28)continue; // Focus on Orange, Red, Light Blue, Pink, Yellow, Dark Blue

        const bPrice=BOARD[bridgeTile].price;
        const interdictBaits=context.isolatedMortgageable.filter(d=>{
          if(d.group===info.groupId)return false;
          if(d.group>=0&&context.groups[d.group].own>=2)return false;
          if(d.group>=0&&(GROUP_TIER[d.group]||1)>info.tier)return false;
          if(d.group>=0){
            const oppCnt=GROUPS[d.group].ids.filter(i=>view.assets[i]?.owner===holderId).length;
            if(oppCnt>=GROUPS[d.group].ids.length-1)return false;
          }
          return Math.abs(d.price-bPrice)<=600;
        }).sort((a,b)=>a.price-b.price);

        let interdicted=false;
        for(const ib of interdictBaits.slice(0,2)){
          if(interdicted)break;
          for(const sweet of [0,80,160]){
            const rawDiff=bPrice-ib.price+sweet;
            const giveCash=rawDiff>0?Math.round(rawDiff/10)*10:0;
            const takeCash=rawDiff<0?Math.min(view.players[holderId]?.cash||0,Math.round(Math.abs(rawDiff)/10)*10):0;
            if(self.cash-giveCash<context.trueReserve+1400)continue;
            const trade={to:holderId,give:[ib.id],take:[bridgeTile],giveCash,takeCash,giveCards:0,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            const oracle=predictPartnerAcceptance(context,trade);
            if(oracle.willAccept&&oracle.myEval.net>=0){
              const action={type:'OFFER',trade,actionId:'nusa-collusion-interdict-'+bridgeTile};
              if(pushCandidate(action,topScore()+305)){
                synthesizedCount++;
                interdicted=true;
                break;
              }
            }
          }
        }
      }
    }

    // Mode C: Stepping-Stone Triangle Trade
    if(synthesizedCount<3&&context.steppingStoneTiles.size>0){
      for(const stepTile of context.steppingStoneTiles){
        if(synthesizedCount>=3)break;
        const thirdOwner=view.assets[stepTile]?.owner;
        if(thirdOwner==null||rejectionActive(view,mem,thirdOwner))continue;
        const candidatesToGive=context.isolatedMortgageable.filter(d=>{
          if(context.steppingStoneTiles.has(d.id))return false;
          if(d.group>=0){
            const cnt=GROUPS[d.group].ids.filter(i=>view.assets[i]?.owner===thirdOwner).length;
            if(cnt>=GROUPS[d.group].ids.length-1)return false;
          }
          return Math.abs(d.price-BOARD[stepTile].price)<=600;
        });
        for(const d of candidatesToGive.slice(0,2)){
          const diff=Math.max(0,BOARD[stepTile].price-d.price);
          for(const sweet of [0,80,160]){
            const trade={to:thirdOwner,give:[d.id],take:[stepTile],giveCash:diff+sweet,takeCash:0,giveCards:0,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            const oracle=predictPartnerAcceptance(context,trade);
            if(oracle.willAccept&&oracle.myEval.net>=0){
              const action={type:'OFFER',trade,actionId:'nusa-stepstone-'+stepTile};
              if(pushCandidate(action,topScore()+260)){
                synthesizedCount++;
                break;
              }
            }
          }
          if(synthesizedCount>=3)break;
        }
      }
    }
  }

  return plan;
}

function compatible(base,action,phase,context){
  if(eq(base,action))return true;
  if(protectedType(base)||protectedType(action))return false;
  if(phase==='buy'){
    const allowedBuy=context.needsPreBuyMortgage
      ?['BUY','AUCTION','MORTGAGE']
      :context.needsPreBuyTrade
        ?['BUY','AUCTION','OFFER']
        :['BUY','AUCTION'];
    return allowedBuy.includes(base.type)&&allowedBuy.includes(action.type);
  }
  if(phase==='auction'){
    const allowedAuction=context.needsMidAuctionMortgage?['BID','PASS','MORTGAGE']:['BID','PASS'];
    return allowedAuction.includes(base.type)&&allowedAuction.includes(action.type);
  }
  if(phase==='trade')return ['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(base.type)&&['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(action.type);
  if(phase==='jail'){
    const allowedJail=['JAIL_PAY','JAIL_CARD','ROLL','BUILD','REDEEM'];
    return allowedJail.includes(base.type)&&allowedJail.includes(action.type);
  }
  if(phase==='buildOffer')return ['BUILD_JOIN','BUILD_DECLINE'].includes(base.type)&&['BUILD_JOIN','BUILD_DECLINE'].includes(action.type);
  if(phase==='mortgage')return ['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(base.type)&&['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(action.type);
  if(phase==='debt')return ['MORTGAGE','SELL','SELL_SET','OFFER'].includes(base.type)&&['MORTGAGE','SELL','SELL_SET','OFFER'].includes(action.type);

  if(phase==='roll'||phase==='end'){
    if(action.type==='SELL'||action.type==='SELL_SET')return false;
    if(action.type==='MORTGAGE'){
      if(BOARD[action.tile]?.type!=='utility'||!context.needsBuildCapital||context.shouldDeferPreRollBuild)return false;
    }
    const allowedTurnTypes=phase==='roll'
      ?(context.shouldDeferPreRollBuild?['ROLL','BUILD','REDEEM']:['ROLL','BUILD','REDEEM','MORTGAGE'])
      :['END','BUILD','REDEEM','MORTGAGE','OFFER'];
    return allowedTurnTypes.includes(base.type)&&allowedTurnTypes.includes(action.type);
  }
  return base.type===action.type;
}

function computeAdvisory(signals,context,salary,action){
  if(!signals.length)return {advisory:0,bonus:0};
  let weightedSum=0,weightTotal=0;
  for(const s of signals){
    const w=weightFor(s.id,context);
    weightedSum+=s.signal*w;
    weightTotal+=w;
  }
  const norm=weightedSum/Math.max(1,Math.sqrt(weightTotal));
  let momentum=0;
  if(norm>0.20){
    if(action.type==='BUILD'||action.type==='BUILD_JOIN')momentum=salary*0.16;
    else if((action.type==='REDEEM'||action.type==='MORTGAGE_REDEEM')&&(context.hasMortgagedMonopoly||context.canDeleverage||context.view.phase==='mortgage'))momentum=salary*0.15;
    else if(action.type==='MORTGAGE'&&(context.needsPreBuyMortgage||context.needsMidAuctionMortgage))momentum=salary*0.19;
    else if(action.type==='OFFER'&&(norm>0.35||context.needsPreBuyTrade))momentum=salary*0.17;
    else if(action.type==='COUNTER'&&norm>0.35)momentum=salary*0.17;
    else if(action.type==='BUY'&&norm>0.25)momentum=salary*0.13;
    else if(action.type==='AUCTION'&&context.earlyCashSinkTiles?.has(context.view.buy))momentum=salary*0.16;
    else if(action.type==='ROLL'&&context.shouldDeferPreRollBuild)momentum=salary*0.18;
  }
  const bonus=Math.max(-salary*0.52,Math.min(salary*0.52,norm*salary*0.28+momentum));
  return {advisory:norm,bonus};
}

export function nusaV7Decision(state,actor,plan,{diagnostics=false,disabled=[]}={}){
  const fallback=(reason,extra={})=>({action:plan?.action??null,changed:false,reason,...extra});
  if(!plan?.action||protectedType(plan.action)||(plan.candidates?.[0]?.score??0)>=1e7)return fallback('protected_or_trivial');
  if(!['auction','buy','roll','end','trade','jail','buildOffer','mortgage','debt'].includes(state.phase))return fallback('protected_phase');

  const context=situationOf(state,actor,plan);
  ensureSovereignCandidates(context,plan);

  const base=plan.action,candidates=plan.candidates||[];

  const salary=context.salary;
  const allowed=new Set(STRATEGY_EXPERTS.map(x=>x.id));
  if(disabled.some(id=>!allowed.has(id)))throw Error('Unknown NUSA expert ablation: '+disabled.find(id=>!allowed.has(id)));

  const eligible=candidates.filter(c=>compatible(base,c.action,state.phase,context));
  if(eligible.length<2)return fallback('no_compatible_alternative');

  const ranked=eligible.map(c=>{
    const signals=runExperts(context,c.action).filter(s=>!disabled.includes(s.id));
    const {advisory,bonus}=computeAdvisory(signals,context,salary,c.action);
    return {...c,bonus,advisory,signals,unified:c.score+bonus};
  }).sort((x,y)=>y.unified-x.unified||x.id.localeCompare(y.id));

  const current=ranked.find(c=>eq(c.action,base));
  if(!current)return fallback('baseline_not_ranked');
  const best=ranked[0];
  const improvement=best.unified-current.unified;

  const changeAllowed=!eq(best.action,base)&&
    improvement>Math.max(5,salary*0.002)&&
    best.score>=current.score-salary*0.92&&
    legalCandidate(context.view,actor,best.action);

  const result={
    action:changeAllowed?best.action:base,
    changed:changeAllowed,
    reason:changeAllowed?'transcendent_metagame_singularity_selected':'native_plan_preserved',
    rankedPool:ranked
  };

  if(diagnostics){
    result.context={
      stage:context.stage,live:context.live,unowned:context.unowned.length,
      bankHouses:context.bank.houses,reserve:Math.round(context.reserve),
      trueReserve:Math.round(context.trueReserve),hotspotReserve:context.hotspotReserve,
      queueAuctionReserve:context.queueAuctionReserve,hotelTrapActive:context.hotelTrapActive,
      contestedAnchors:context.contestedAnchorTiles.size,earlyCashSinks:context.earlyCashSinkTiles.size,
      splitThreeWayGroups:context.splitThreeWayGroups.length,collusionTargets:context.collusionTargetTiles.size,
      repairSafe:context.cardCycle.repairSafe,
      winProbability:+context.winProbability.toFixed(3),isCriticalTurn:context.isCriticalTurn
    };
    result.proposals=ranked.slice(0,8).map(c=>({id:c.id,type:c.action.type,score:+c.score.toFixed(2),unified:+c.unified.toFixed(2)}));
    result.expertCount=STRATEGY_EXPERTS.length;
    result.candidateCount=candidates.length;
    result.eligibleCount=eligible.length;
  }
  return result;
}

function comparableCandidate(first,cand,phase){
  if(eq(first,cand))return true;
  if(isBuild(first))return isBuild(cand);
  if(first.type==='OFFER')return cand.type==='OFFER'||isBuild(cand)||cand.type==='MORTGAGE'||(phase==='buy'&&cand.type==='AUCTION');
  if(phase==='buy')return ['BUY','AUCTION','MORTGAGE','OFFER'].includes(cand.type);
  if(phase==='auction')return ['BID','PASS','MORTGAGE'].includes(cand.type);
  if(phase==='buildOffer')return ['BUILD_JOIN','BUILD_DECLINE'].includes(cand.type);
  if(phase==='mortgage')return ['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(cand.type);
  if(phase==='trade')return ['TRADE_ACCEPT','COUNTER','TRADE_REJECT'].includes(cand.type);
  if(phase==='jail')return ['JAIL_PAY','JAIL_CARD','ROLL','BUILD','REDEEM'].includes(cand.type);
  return first.type===cand.type;
}

function searchCandidatePool(plan,council,phase,maxRoots){
  const first=council.action;
  const source=council.rankedPool||plan.candidates;
  const selected=source.filter(c=>comparableCandidate(first,c.action,phase));
  const base=selected.find(c=>eq(c.action,first));
  if(!base)return [];

  return [
    base,

      .sort((a,b)=>(b.unified??b.score)-(a.unified??a.score)||b.score-a.score||a.id.localeCompare(b.id))
  ].slice(0,maxRoots);
}

function sovereignLeafValue(s,actor){
  const posVal=positionValue(s,actor,{noRisk:false,noLearning:true});
  if(s.players[actor]?.out||s.phase==='won')return posVal;
  const myWorth=worth(s,actor);
  const aliveOthers=s.players.filter(p=>p.id!==actor&&!p.out);
  const maxRivalWorth=Math.max(1,...aliveOthers.map(p=>worth(s,p.id)));
  const worthEdge=(myWorth-maxRivalWorth*0.85)*0.28;

  let monopolyBonus=0;
  for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
    if(complete(s,actor,gIdx)){
      const g=GROUPS[gIdx];
      const tier=GROUP_TIER[gIdx]||1;
      const minLv=Math.min(...g.ids.map(i=>s.assets[i]?.level||0));
      monopolyBonus+=(420+Math.min(3,minLv)*260)*tier;
    }
  }
  return posVal*0.74+worthEdge+monopolyBonus;
}

function omniscientRolloutStep(s,who,rng,rootActor){
  if(who===rootActor){
    const p=s.players[who];
    if(s.phase==='buildOffer'&&s.buildOffer?.pending?.includes(who)){
      for(const g of GROUPS){

          if(legalCandidate(s,who,joinAct))return joinAct;
        }
      }
    }
    if(s.phase==='buy'&&s.turn===who&&Number.isInteger(s.buy)){

      const isCashSink=t?.group===6&&GROUPS[6].ids.every(i=>s.assets[i]?.owner==null)&&(p.cash-price<2600);
      if(!isCashSink&&p.cash>=price){

        if(legalCandidate(s,who,buyAct))return buyAct;
      }
    }
    if(s.phase==='auction'&&s.auction&&s.auction.participants?.includes(who)&&s.auction.highest!==who){

          if(legalCandidate(s,who,bidAct))return bidAct;
        }
      }else{
        const t=BOARD[s.auction.tile];
        if(t?.price){

          const minBid=Math.max(100,(s.auction.amount||0)+10);

        }
      }

function evaluateWorld(publicView,actor,candidate,scenarioSeed,opts,expired){
  const random=new ScenarioRandom(scenarioSeed);
  let s=sampleWorld(publicView,random.stream(0,0,'sampled-decks'));
  const initial=sovereignLeafValue(s,actor);
  let nodes=0;
  try{
    const root=command(s,actor,candidate.action,random.forAction(s,actor,candidate.action));
    nodes++;s=root;
    if(s.log?.length>80)s.log=s.log.slice(-80);
    const path=Array.isArray(candidate.macro)?candidate.macro.slice(1,6):[];
    for(const part of path){
      if(expired()||simulationActor(s)!==actor||!legalCandidate(s,actor,part))break;
      s=command(s,actor,part,random.forAction(s,actor,part));nodes++;
    }
    const horizon=publicView.turnSerial+opts.horizonTurns;
    const maxEngineSteps=Math.max(opts.horizonSteps,opts.horizonTurns*12);
    for(let step=0;step<maxEngineSteps&&s.phase!=='won';step++){
      if(s.turnSerial>=horizon&&['roll','jail','won'].includes(s.phase))break;
      if(expired())return {ok:false,nodes,reason:'budget'};
      const who=simulationActor(s),action=omniscientRolloutStep(s,who,random.policy(s,who),actor);
      if(!action)break;
      s=command(s,who,action,random.forAction(s,who,action));nodes++;
      if(s.log?.length>80)s.log=s.log.slice(-80);
    }
    return {ok:true,nodes,value:sovereignLeafValue(s,actor)-initial};
  }catch{
    return {ok:false,nodes,reason:'sandbox_exception'};
  }
}

  };
}

export function choosePairedSearchEvidence(pool,returns,salary){
  if(pool.length<2||returns.length!==pool.length||!returns[0]?.length)
    return {selectedIndex:0,bestDelta:0};
  const baseScore=pool[0].unified??pool[0].score;
  const utilities=pool.map((c,i)=>{
    const rs=returns[i];
    if(!rs||!rs.length)return -Infinity;
    const mean=rs.reduce((a,b)=>a+b,0)/rs.length;
    const sorted=rs.slice().sort((a,b)=>a-b);
    const tailCount=Math.max(1,Math.ceil(sorted.length*0.25));
    const cvar=sorted.slice(0,tailCount).reduce((a,b)=>a+b,0)/tailCount;
    const blended=mean*0.72+cvar*0.28;
    return ((c.unified??c.score)-baseScore)+blended*0.38;
  });
  const order=pool.map((_,i)=>i).sort((a,b)=>utilities[b]-utilities[a]);
  const best=order[0],delta=utilities[best]-utilities[0];
  const bestReturns=returns[best],ctrlReturns=returns[0];
  const commonLen=Math.min(bestReturns.length,ctrlReturns.length);
  const pairs=bestReturns.slice(0,commonLen).map((v,i)=>v-ctrlReturns[i]);
  const wins=pairs.filter(d=>d>0).length;
  const enough=best!==0&&commonLen>=2&&wins>=Math.ceil(commonLen*0.65)&&delta>Math.max(25,salary*0.012);
  return {selectedIndex:enough?best:0,bestDelta:+delta.toFixed(2)};
}

import {legalCandidate} from './candidates.mjs';
import {decideNusa,NUSA_MODEL_VERSION} from './nusa-v7/index.mjs';
import {cleanAction} from './nusa-v7/council.mjs';
import {apexPolicyAction,APEX_ADAPTER_VERSION} from './apex-policy.mjs';
import {kresnaPolicyAction,KRESNA_MODEL_VERSION} from './kresna-policy.mjs';

export const STRATEGIC_PERSONAS=Object.freeze(['nusa','kresna','apex']);

export function decidePersona(state,actor,options={}){
  const persona=state?.players?.[actor]?.botPersona;
  if(!STRATEGIC_PERSONAS.includes(persona))return decideMind(state,actor,options);
  const started=performance.now();
  const view=observation(state,actor);
  const base=planDecision(view,actor,{fast:false,noLearning:true});
  const modelLabel=persona==='nusa'?'NUSA-V18.0-SINGULARITY':persona==='kresna'?'KRESNA-V2.0':'APEX-JS-3.0';
  if(!base.action)return {...base,diagnostics:{model:modelLabel,algorithm:'no_action',persona}};

  let selected;
  try{
    if(persona==='nusa'){
      const budgetMs=Number.isFinite(options.timeBudgetMs)?options.timeBudgetMs:(options.deep?2600:64);
      const maxNodes=options.deep||budgetMs>=1500?2800:220;
      selected=decideNusa(view,actor,base,{
        mode:options.nusaMode||'search',
        maxMs:budgetMs,
        maxNodes,
        deep:!!options.deep||budgetMs>=1500
      });
    }else if(persona==='kresna'){
      selected=kresnaPolicyAction(view,actor,base);
    }else{
      selected=apexPolicyAction(view,actor,base);
    }
  }catch{
    selected={action:base.action,changed:false,reason:'fallback'};
  }

  const proposed=selected?.action;
  const matched=base.candidates.find(c=>JSON.stringify(cleanAction(c.action))===JSON.stringify(cleanAction(proposed)));
  const candidateAction=matched?matched.action:proposed;
  const valid=!!candidateAction&&legalCandidate(view,actor,candidateAction);
  const action=valid?candidateAction:base.action;

  return {
    ...base, action, shouldConsult:false,
    diagnostics:{
      version:persona==='nusa'?NUSA_MODEL_VERSION:persona==='kresna'?KRESNA_MODEL_VERSION:APEX_ADAPTER_VERSION,
      model:modelLabel, algorithm:'transcendent-metagame-singularity-v18.0', persona,
      selectedBy:valid&&selected.changed?'strategic-persona':'economic-floor',
      reason:valid?selected.reason:'illegal_proposal_rejected',
      actor, elapsedMs:performance.now()-started, nodes:selected?.search?.nodes??0, samples:selected?.search?.pairedSamples??0, learning:{enabled:false}, selected:action.actionId, search:[]
    }
  };
}
`;

import {createGame,command,decisionActor,worth} from '../lib/game.mjs';
import {decidePersona} from '../lib/bot/persona-router.mjs';

const args=process.argv.slice(2);
const numGames=Number(args.find(a=>!a.startsWith('-'))||6);
const maxSteps=Number(args.filter(a=>!a.startsWith('-'))[1]||500);
const deepMode=args.includes('--deep');

const BASE_ROSTER=[
  {name:'NUSA',controller:'bot',botPersona:'nusa',botDifficulty:'hardcore'},
  {name:'Kresna',controller:'bot',botPersona:'kresna',botDifficulty:'hardcore'},
  {name:'Apex',controller:'bot',botPersona:'apex',botDifficulty:'hardcore'},
  {name:'Raka',controller:'bot',botPersona:'raka',botDifficulty:'hardcore'},
  {name:'Bima',controller:'bot',botPersona:'bima',botDifficulty:'hardcore'},
  {name:'Sari',controller:'bot',botPersona:'sari',botDifficulty:'hardcore'}
];

const stats=Object.fromEntries(BASE_ROSTER.map(r=>[r.botPersona.toUpperCase(),{persona:r.botPersona.toUpperCase(),wins:0,survived:0,totalWorth:0,points:0}]));
const F1_POINTS=[6,5,4,3,2,1];

console.log('=== ARENA BENCHMARK 6-MODEL (NUSA V18.0 TRANSCENDENT SINGULARITY) ===');
console.log('Mode Berpikir NUSA : '+(deepMode?'Gear 3 Deep Chronos (2.000-3.500 ms pada titik kritis)':'Gear 2 Tactical Turbo (90-180 ms)'));
console.log('Menjalankan '+numGames+' pertandingan rotasi kursi adil (maks '+maxSteps+' langkah/game)...\\n');

for(let g=0;g<numGames;g++){
  let seed=(20261018+g*9973)>>>0;
  const rng=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296;};
  const rotated=BASE_ROSTER.map((_,i)=>BASE_ROSTER[(i+g)%BASE_ROSTER.length]);
  let s=createGame({players:rotated,startingRoll:false},rng);
  let steps=0;
  while(s.phase!=='won'&&steps<maxSteps){
    const actor=decisionActor(s);
    const p=s.players[actor];
    const isNusa=p?.botPersona==='nusa';
    const plan=decidePersona(s,actor,{
      nusaMode:'search',
      difficulty:'hardcore',
      timeBudgetMs:isNusa?(deepMode?2400:110):32,
      deep:isNusa&&deepMode
    });
    if(!plan?.action)break;
    s=command(s,actor,plan.action,rng);
    steps++;
  }
  const ranked=s.players.map(p=>({
    persona:p.botPersona.toUpperCase(),
    out:p.out,
    worth:p.out?0:Math.round(worth(s,p.id)/10)
  })).sort((a,b)=>(Number(!b.out)-Number(!a.out))||(b.worth-a.worth));

  const winner=ranked[0];
  stats[winner.persona].wins++;
  ranked.forEach((r,idx)=>{
    if(!r.out)stats[r.persona].survived++;
    stats[r.persona].totalWorth+=r.worth;
    stats[r.persona].points+=F1_POINTS[idx]||0;
  });
  console.log('  Match #'+String(g+1).padStart(2,'0')+': Winner = '+winner.persona.padEnd(7)+' (steps='+steps+', worth=N '+winner.worth+')');
}

const table=Object.values(stats)
  .map(x=>({
    persona:x.persona,
    wins:x.wins,
    winRate:((x.wins/numGames)*100).toFixed(1)+'%',
    survivalRate:((x.survived/numGames)*100).toFixed(1)+'%',
    avgWorth:Math.round(x.totalWorth/numGames),
    points:x.points
  }))
  .sort((a,b)=>b.wins-a.wins||b.points-a.points||b.avgWorth-a.avgWorth);

console.log('\\n=== HASIL AKHIR KLASEMEN ('+numGames+' PERTANDINGAN) ===');
console.table(table);
`;
