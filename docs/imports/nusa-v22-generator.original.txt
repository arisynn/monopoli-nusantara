import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';

const ROOT = process.cwd();

function ensureDir(filePath) {
  fs.mkdirSync(path.dirname(filePath), { recursive: true });
}

function writeText(relPath, content) {
  const full = path.join(ROOT, relPath);
  ensureDir(full);
  fs.writeFileSync(full, content.trimStart(), 'utf8');
  console.log('  [UPDATE] ' + relPath);
}

function sha256File(relPath) {
  const buf = fs.readFileSync(path.join(ROOT, relPath));
  return crypto.createHash('sha256').update(buf).digest('hex');
}

function toLibBotImports(labCode) {
  return labCode
    .replaceAll("'../../lib/content.mjs'", "'../../content.mjs'")
    .replaceAll("'../../lib/game.mjs'", "'../../game.mjs'")
    .replaceAll("'../../lib/bot/", "'../");
}

function listFilesRecursive(dirRel) {
  const results = [];
  const absDir = path.join(ROOT, dirRel);
  if (!fs.existsSync(absDir)) return results;
  for (const entry of fs.readdirSync(absDir, { withFileTypes: true })) {
    const rel = path.posix.join(dirRel.replaceAll('\\', '/'), entry.name);
    if (entry.isDirectory()) results.push(...listFilesRecursive(rel));
    else results.push(rel);
  }
  return results.sort();
}

const LAB_BOARD_GRAPH_MJS = `
/** NUSA V22.0 "Omniscient Cortex" — Hypergraph Board Reader & Multi-Agent Strategic Planner.
 * Reads the entire board and all 6 players to construct:
 * 1. Group Ownership Topology (\`groupTopology\`)
 * 2. 6-Player Desire Matrix (\`desireMatrix[playerId][tileId]\` on 0..100 scale)
 * 3. Trojan-Bait Inventory (\`trojanBaitTiles\` — pairs where NUSA holds the 3rd blocker)
 * 4. 1-Hop & 2-Hop Broker Acquisition Plans (\`brokerPlans\`, \`brokerTargetTiles\`)
 * 5. 3-Layer Active Monopoly Interdiction (\`interdictionTargets\`, \`poisonPillTiles\`)
 * 6. Spatial-Temporal Pawn Corridor Radar (\`corridorRadar\`)
 */
import {BOARD,GROUPS} from '../../lib/content.mjs';
import {rent} from '../../lib/game.mjs';

export const GROUP_TIER=Object.freeze([1.14,1.48,1.30,1.72,1.62,1.32,0.84,1.36]);
export const STATION_IDS=Object.freeze([5,15,25,35]);
export const UTILITY_IDS=Object.freeze([12,28]);
const DICE_PROB=Object.freeze([0,0,1/36,2/36,3/36,4/36,5/36,6/36,5/36,4/36,3/36,2/36,1/36]);

export function analyzeBoardGraph(view,actor,opponentProfiles=new Map()){
  const self=view.players[actor];
  const alivePlayers=view.players.filter(p=>!p.out);
  const rivals=alivePlayers.filter(p=>p.id!==actor);
  const ownerOf=tid=>view.assets[tid]?.owner;

  // 1. SPATIAL-TEMPORAL PAWN CORRIDOR RADAR
  const tileRivalRollProb=Array(40).fill(0);
  const tileSelfRollProb=Array(40).fill(0);

  for(let d=2;d<=12;d++){
    const p=DICE_PROB[d]||0;
    const selfBase=self.inJail?10:self.pos;
    tileSelfRollProb[(selfBase+d)%40]+=p*(self.inJail?0.45:1.0);
  }

  for(const r of rivals){
    const rBase=r.inJail?10:r.pos;
    const weight=r.inJail?0.55:1.0;
    for(let d=2;d<=12;d++){
      const dest=(rBase+d)%40;
      tileRivalRollProb[dest]+=(DICE_PROB[d]||0)*weight;
    }
  }

  let selfImmediateDangerExpected=0;
  let selfMaxSingleHit=0;
  let unownedStrategicAhead=0;
  for(let d=2;d<=12;d++){
    const dest=((self.inJail?10:self.pos)+d)%40;
    const b=BOARD[dest];
    const a=view.assets[dest];
    const prob=DICE_PROB[d]||0;
    if(b?.price&&a?.owner!=null&&a.owner!==actor&&!a.mortgaged){
      const rVal=rent(view,dest,d);
      selfImmediateDangerExpected+=prob*rVal;
      if(prob>=1/18)selfMaxSingleHit=Math.max(selfMaxSingleHit,rVal);
    }else if(b?.price&&a?.owner==null){
      if(b.type==='station'||(b.group>=0&&(GROUP_TIER[b.group]||1)>=1.25)){
        unownedStrategicAhead+=prob*b.price;
      }
    }
  }

  const spatialDangerFactor=selfMaxSingleHit>=1800||selfImmediateDangerExpected>=380
    ?1.26
    :selfMaxSingleHit<=350&&selfImmediateDangerExpected<=95
      ?0.68
      :0.95;

  const groupAmbushIntensity=GROUPS.map((g,gIdx)=>{
    const sumProb=g.ids.reduce((acc,tid)=>acc+(tileRivalRollProb[tid]||0),0);
    return +(sumProb*(GROUP_TIER[gIdx]||1)).toFixed(4);
  });

  // 2. 6-PLAYER DESIRE MATRIX & TROJAN-BAIT DETECTION
  const desireMatrix=new Map();
  for(const p of alivePlayers){
    const scores=Array(40).fill(0);
    const prof=opponentProfiles.get(p.id);
    const psychGroups=prof?.desiredGroups||{};
    const persona=p.botPersona||'human';

    for(const t of BOARD){
      if(!t.price)continue;
      const tid=t.id;
      if(ownerOf(tid)===p.id)continue;

      if(t.type==='station'){
        const stOwned=STATION_IDS.filter(i=>ownerOf(i)===p.id).length;
        const personaBoost=persona==='raka'?18:0;
        scores[tid]=Math.min(92,28+stOwned*20+personaBoost+(tid===5?8:0));
        continue;
      }
      if(t.type==='utility'){
        const utOwned=UTILITY_IDS.filter(i=>ownerOf(i)===p.id).length;
        scores[tid]=utOwned>=1?48:15;
        continue;
      }
      if(t.group>=0){
        const gIds=GROUPS[t.group].ids;
        const tier=GROUP_TIER[t.group]||1;
        const pCount=gIds.filter(i=>ownerOf(i)===p.id).length;
        const otherOwners=new Set(gIds.filter(i=>i!==tid&&ownerOf(i)!=null&&ownerOf(i)!==p.id).map(i=>ownerOf(i)));
        const psychBonus=Math.min(16,(psychGroups[t.group]||0)*6);

        if(pCount===gIds.length-1){
          scores[tid]=Math.min(100,Math.round(88+tier*7+psychBonus));
        }else if(pCount===1&&gIds.length===3){
          if(otherOwners.size===0){
            scores[tid]=Math.min(86,Math.round(62*tier+psychBonus));
          }else{
            scores[tid]=Math.min(78,Math.round(48*tier+psychBonus));
          }
        }else if(pCount===0&&otherOwners.size===0){
          scores[tid]=Math.round(24*tier+psychBonus);
        }else{
          scores[tid]=Math.round(14*tier+psychBonus);
        }
      }
    }
    desireMatrix.set(p.id,scores);
  }

  // Trojan Baits: Tiles owned by NUSA where giving the tile to Rival R gives R 2/3 of a 3-tile group,
  // while NUSA ALSO owns the 3rd tile (or can keep the 3rd tile as an unbreakable hostage), OR where
  // Rival R strongly desires the tile (desire >= 52) without completing R's monopoly!
  const trojanBaitForRival=new Map();
  for(const r of rivals){
    const rDesire=desireMatrix.get(r.id)||[];
    const list=[];
    for(const t of BOARD){
      if(!t.price||ownerOf(t.id)!==actor||(view.assets[t.id]?.level||0)>0)continue;
      if(t.group>=0){
        const gIds=GROUPS[t.group].ids;
        if(gIds.some(i=>(view.assets[i]?.level||0)>0))continue;
        const rCount=gIds.filter(i=>ownerOf(i)===r.id).length;
        const myCount=gIds.filter(i=>ownerOf(i)===actor).length;
        // Never hand over a tile that completes R's monopoly!
        if(rCount>=gIds.length-1)continue;
        // Never break our own 2/3 near-monopoly in a high-tier group unless we are keeping a hostage in low-tier
        if(myCount>=2&&(GROUP_TIER[t.group]||1)>=1.28)continue;

        const isLockedHostagePair=(gIds.length===3&&rCount===1&&myCount===2&&(GROUP_TIER[t.group]||1)<1.28);
        const appeal=rDesire[t.id]+(isLockedHostagePair?24:0);
        list.push({tileId:t.id,appeal,isLockedHostagePair,price:t.price});
      }else if(t.type==='station'||t.type==='utility'){
        const mySt=STATION_IDS.filter(i=>ownerOf(i)===actor).length;
        if(t.type==='station'&&mySt>=3)continue;
        list.push({tileId:t.id,appeal:rDesire[t.id],isLockedHostagePair:false,price:t.price});
      }
    }
    list.sort((a,b)=>b.appeal-a.appeal||a.price-b.price);
    trojanBaitForRival.set(r.id,list);
  }

  // 3. GROUP OWNERSHIP TOPOLOGY & INTERDICTION / BROKER TARGETS
  const poisonPillTiles=new Set();       // Unowned tiles we MUST buy to plant a blocking anchor
  const unownedCriticalDeny=new Set();   // Unowned tiles that would complete a rival's monopoly
  const thirdPartyInterceptions=[];      // Tiles held by Rival B that Rival A (2/3) is about to buy
  const thirdPartyInterceptionTiles=new Set();
  const brokerPlans=[];                  // 1-Hop and 2-Hop acquisition plans for NUSA's near-monopolies
  const brokerTargetTiles=new Set();     // Tiles held by B (or unowned) that A wants, so we can trade to A for our key tile!
  const brokerTileToPlan=new Map();
  const rivalCashStarvationTargets=new Map();

  const groupTopologies=GROUPS.map((g,gIdx)=>{
    const ids=g.ids;
    const tier=GROUP_TIER[gIdx]||1;
    const houseCost=BOARD[ids[0]]?.build||1000;
    const myTiles=ids.filter(i=>ownerOf(i)===actor);
    const unownedTiles=ids.filter(i=>ownerOf(i)==null);
    const rivalCounts=rivals.map(r=>({
      id:r.id,
      cash:r.cash,
      persona:r.botPersona||'human',
      tiles:ids.filter(i=>ownerOf(i)===r.id)
    })).filter(x=>x.tiles.length>0).sort((a,b)=>b.tiles.length-a.tiles.length||b.cash-a.cash);

    let status='OPEN';
    if(myTiles.length===ids.length){
      status='MONOPOLY_OURS';
    }else if(rivalCounts[0]?.tiles.length===ids.length){
      status='MONOPOLY_RIVAL';
      const rOwner=rivalCounts[0];
      const maxLv=Math.max(...ids.map(i=>view.assets[i]?.level||0));
      if(maxLv<3){
        rivalCashStarvationTargets.set(rOwner.id,{
          rivalId:rOwner.id,
          groupId:gIdx,
          dangerCashThreshold:houseCost*ids.length,
          priority:tier*1.35
        });
      }
    }else if(myTiles.length===ids.length-1){
      status='OUR_NEAR_MONOPOLY';
      const missingTile=ids.find(i=>ownerOf(i)!==actor);
      const holderId=ownerOf(missingTile);
      if(holderId!=null&&holderId!==actor){
        const holderDesire=desireMatrix.get(holderId)||[];
        const myBaits=trojanBaitForRival.get(holderId)||[];

        // Scan 2-Hop Broker Baits: Tiles Y (unowned or owned by third-party B) that holderId strongly desires
        const twoHopBaits=[];
        for(const candT of BOARD){
          if(!candT.price||candT.id===missingTile)continue;
          const yOwner=ownerOf(candT.id);
          if(yOwner===actor||yOwner===holderId)continue;
          if((view.assets[candT.id]?.level||0)>0)continue;
          if(candT.group>=0&&GROUPS[candT.group].ids.some(i=>(view.assets[i]?.level||0)>0))continue;
          // Ensure giving candT to holderId does NOT complete holderId's monopoly!
          if(candT.group>=0){
            const hCount=GROUPS[candT.group].ids.filter(i=>ownerOf(i)===holderId).length;
            if(hCount>=GROUPS[candT.group].ids.length-1)continue;
          }
          // If owned by third-party B, ensure B isn't holding a 2/3 pair on candT
          if(yOwner!=null&&candT.group>=0){
            const bCount=GROUPS[candT.group].ids.filter(i=>ownerOf(i)===yOwner).length;
            if(bCount>=2)continue;
          }
          const desireByKeyHolder=holderDesire[candT.id]||0;
          if(desireByKeyHolder>=54){
            twoHopBaits.push({
              baitTile:candT.id,
              baitOwner:yOwner, // null if unowned, or Rival B's id
              desireScore:desireByKeyHolder,
              ultimateTargetTile:missingTile,
              keyHolderId:holderId,
              targetGroupId:gIdx,
              priority:tier*(desireByKeyHolder/60)
            });
          }
        }
        twoHopBaits.sort((a,b)=>b.priority-a.priority);
        for(const tb of twoHopBaits.slice(0,4)){
          brokerTargetTiles.add(tb.baitTile);
          if(!brokerTileToPlan.has(tb.baitTile)){
            brokerTileToPlan.set(tb.baitTile,tb);
          }
        }

        brokerPlans.push({
          groupId:gIdx,
          tier,
          houseCost,
          missingTile,
          keyHolderId:holderId,
          directBaits:myBaits.slice(0,4),
          twoHopBaits:twoHopBaits.slice(0,4)
        });
      }
    }else if(rivalCounts[0]?.tiles.length===ids.length-1){
      status='RIVAL_NEAR_MONOPOLY';
      const dangerRival=rivalCounts[0];
      const missingTile=ids.find(i=>ownerOf(i)!==dangerRival.id);
      const holderId=ownerOf(missingTile);

      rivalCashStarvationTargets.set(dangerRival.id,{
        rivalId:dangerRival.id,
        groupId:gIdx,
        dangerCashThreshold:houseCost*ids.length+BOARD[missingTile]?.price,
        priority:tier*1.25
      });

      if(holderId==null){
        unownedCriticalDeny.add(missingTile);
        poisonPillTiles.add(missingTile);
      }else if(holderId!==actor){
        // THIRD-PARTY LEAK EMERGENCY: Rival A has 2/3, Rival B holds the 3rd tile!
        if((view.assets[missingTile]?.level||0)===0){
          thirdPartyInterceptionTiles.add(missingTile);
          thirdPartyInterceptions.push({
            tileId:missingTile,
            groupId:gIdx,
            tier,
            dangerRivalId:dangerRival.id,
            holderRivalId:holderId,
            priority:tier*(dangerRival.cash>=houseCost*2?1.45:1.15)
          });
        }
      }
    }else if(ids.length===3&&unownedTiles.length===0&&myTiles.length===1&&rivalCounts.length===2){
      status='SPLIT_1_1_1';
    }else if(myTiles.length===0&&rivalCounts.length>=1&&unownedTiles.length>=1&&tier>=1.14){
      status='RIVAL_SEEDED_OPEN';
      for(const ut of unownedTiles){
        poisonPillTiles.add(ut);
      }
    }

    return {
      groupId:gIdx,
      ids,
      tier,
      houseCost,
      status,
      myCount:myTiles.length,
      unownedCount:unownedTiles.length,
      rivalCounts,
      ambushIntensity:groupAmbushIntensity[gIdx]
    };
  });

  brokerPlans.sort((a,b)=>b.tier-a.tier);
  thirdPartyInterceptions.sort((a,b)=>b.priority-a.priority);

  // Count how many rival groups NUSA currently holds a blocking hostage in
  const hostageGroupsHeld=groupTopologies.filter(gt=>
    gt.myCount>=1&&gt.myCount<gt.ids.length&&gt.rivalCounts.some(r=>r.tiles.length>=1)
  ).length;

  return Object.freeze({
    groupTopologies,
    desireMatrix,
    trojanBaitForRival,
    brokerPlans,
    brokerTargetTiles,
    brokerTileToPlan,
    poisonPillTiles,
    unownedCriticalDeny,
    thirdPartyInterceptions,
    thirdPartyInterceptionTiles,
    rivalCashStarvationTargets,
    hostageGroupsHeld,
    corridorRadar:Object.freeze({
      tileRivalRollProb,
      tileSelfRollProb,
      groupAmbushIntensity,
      selfImmediateDangerExpected,
      selfMaxSingleHit,
      unownedStrategicAhead,
      spatialDangerFactor
    })
  });
}

export function getAuctionQuantum(view,actor=view?.auction?.bidder){
  const cur=view?.auction?.amount||0;
  const cfgInc=Number(AUCTION_RULES?.increment)||50;
  const cfgMin=Number(AUCTION_RULES?.minimum)||100;

  if(view?.phase==='auction'&&Number.isInteger(actor)&&view.players?.[actor]){
    const probeView={
      ...view,
      players:view.players.map((p,i)=>i===actor?{...p,cash:Math.max(p.cash,500000)}:p)
    };
    const stepCandidates=cfgInc!==10
      ?[...new Set([cfgInc,50,5,10,25,100])]
      :[5,10,25,50,100];

    const minCandidates=[...new Set([
      Math.max(cfgMin,cur+cfgInc),
      ...stepCandidates.map(s=>Math.max(cfgMin,Math.ceil((cur+s)/s)*s)),
      ...stepCandidates.map(s=>Math.ceil((cur+s)/s)*s)
    ])].filter(x=>x>cur).sort((a,b)=>a-b);

    for(const minBid of minCandidates){
      if(legalCandidate(probeView,actor,{type:'BID',amount:minBid})){
        for(const step of stepCandidates){
          if(minBid%step===0&&legalCandidate(probeView,actor,{type:'BID',amount:minBid+step})){
            return {step,minBid};
          }
        }
        return {step:cfgInc,minBid};
      }
    }
  }

  const step=cfgInc;
  const rawMin=Math.max(cfgMin,cur+step);
  const minBid=Math.ceil(rawMin/step)*step;
  return {step,minBid};
}

export function analyzePublicCardCycles(view){
  const held=new Set((view.players||[]).flatMap(p=>p.cards||[]).map(c=>c.deck+':'+c.idx));
  const recentChance=[];
  const recentCommunity=[];
  const seenChance=new Set();
  const seenCommunity=new Set();

  for(const ev of (view.log||[])){
    const c=ev?.card||(ev?.finance?.[0]?.card);
    if(!c||!Number.isInteger(c.idx))continue;
    if(c.deck==='chance'&&!held.has('chance:'+c.idx)&&!seenChance.has(c.idx)){
      seenChance.add(c.idx);
      recentChance.push(c.idx);
    }else if(c.deck==='community'&&!held.has('community:'+c.idx)&&!seenCommunity.has(c.idx)){
      seenCommunity.add(c.idx);
      recentCommunity.push(c.idx);
    }
  }

  const bottomChance=new Set(recentChance.slice(0,12));
  const bottomCommunity=new Set(recentCommunity.slice(0,12));
  const chanceRepairSafe=bottomChance.has(11);
  const commRepairSafe=bottomCommunity.has(13);
  const repairSafe=chanceRepairSafe&&commRepairSafe;
  const repairImminent=(!chanceRepairSafe&&recentChance.length>=9)||(!commRepairSafe&&recentCommunity.length>=9);

  const activeMagnetTiles=new Set();
  if(!bottomChance.has(0))activeMagnetTiles.add(39);
  if(!bottomChance.has(2))activeMagnetTiles.add(24);
  if(!bottomChance.has(3))activeMagnetTiles.add(11);
  if(!bottomChance.has(9))activeMagnetTiles.add(19);
  if(!bottomChance.has(13))activeMagnetTiles.add(5);
  if(!bottomChance.has(4)||!bottomChance.has(5)){
    activeMagnetTiles.add(5);
    activeMagnetTiles.add(15);
    activeMagnetTiles.add(25);
  }

  return Object.freeze({
    repairSafe,
    repairImminent,
    activeMagnetTiles,
    recentChanceCount:recentChance.length,
    recentCommunityCount:recentCommunity.length
  });
}

export function situationOf(state,actor,plan){
  const view=observation(state,actor),self=view.players[actor];
  if(!self||self.out)throw Error('NUSA V22.0 actor inactive');
  const rivals=view.players.filter(p=>p.id!==actor&&!p.out);
  const owned=BOARD.filter(t=>t.price&&view.assets[t.id]?.owner===actor);
  const unowned=BOARD.filter(t=>t.price&&view.assets[t.id]?.owner==null);
  const bank=stock(view);
  const cardCycle=analyzePublicCardCycles(view);
  const {step:auctionStep,minBid:auctionMinBid}=getAuctionQuantum(view,actor);

  const rawOppMem=view.botMemory?.opponents||{};

export function completion(c,t,who=c.actor){
  const b=BOARD[t];

  if(b.type==='station'){

    return ownCount>=1?0.40:0.10;
  }
  if(b.group<0)return 0;
  const ids=GROUPS[b.group].ids;

export function block(c,t){
  const b=BOARD[t];
  if(!b)return 0;
  if(c.boardGraph?.unownedCriticalDeny?.has(t))return 0.99;
  if(c.boardGraph?.thirdPartyInterceptionTiles?.has(t))return 0.96;
  if(b.type==='station'){
    const rivalMax=Math.max(0,...c.rivals.map(p=>[5,15,25,35].filter(i=>owner(c,i)===p.id&&i!==t).length));
    return rivalMax>=2?0.72:rivalMax===1?0.35:0;
  }
  if(b.group<0)return 0;
  const g=c.groups[b.group];
  const tier=GROUP_TIER[b.group]||1;
  let maxThreat=0;
  for(const o of g.owners){
    const isPriorityRival=(o.persona==='bima'||o.persona==='kresna')||(c.leader?.id===o.id);
    if(o.count===g.ids.length-1){
      const rivalCash=c.view.players[o.id]?.cash||0;
      const readiness=clamp(rivalCash/Math.max(1000,g.ids.length*2*g.houseCost),0.35,1.25);
      const boost=isPriorityRival?1.30:1.0;
      maxThreat=Math.max(maxThreat,clamp(0.95*readiness*boost,0,1));

      const anchorVal=tier>=1.25?(isPriorityRival?0.80:0.70):tier>=1.10?0.55:0.30;
      maxThreat=Math.max(maxThreat,anchorVal);
    }
  }
  return maxThreat;
}

    const fifoBoost=c.cardCycle?.activeMagnetTiles?.has(t)?p.price*0.14:0;
    const brokerBonus=c.boardGraph?.brokerTargetTiles?.has(t)?p.price*0.30:0;
    return p.price*(1.05+ownSt*0.36)*geoMult+fifoBoost+brokerBonus;
  }
  const tier=p.group>=0?(GROUP_TIER[p.group]||1):1;
  const synergy=completion(c,t,c.actor);
  const denial=block(c,t);
  const apexBonus=APEX_TILES.has(t)?p.price*0.08*tier:0;
  const poisonBonus=c.boardGraph?.poisonPillTiles?.has(t)?p.price*0.30*tier:0;
  const brokerBonus=c.boardGraph?.brokerTargetTiles?.has(t)?p.price*0.38*tier:0;
  const interceptBonus=c.boardGraph?.thirdPartyInterceptionTiles?.has(t)?p.price*0.42*tier:0;
  const chainBonus=c.chainTargetTiles?.has(t)?p.price*0.24*tier:0;
  const visits=(c.visits[t]||0)+0.65*(c.immediateVisits?.[t]||0);
  const magnetBonus=c.cardCycle?.activeMagnetTiles?.has(t)?p.price*0.20:0;
  const rent3=p.rents?.[3]||p.rents?.[0]||200;
  return p.price*tier*(0.88+synergy*1.18+denial*0.74)+visits*rent3*0.22+magnetBonus+apexBonus+poisonBonus+brokerBonus+interceptBonus+chainBonus;
}

export function grandmasterAuctionCeiling(c){
  const auction=c.view.auction;
  if(!auction)return 0;
  const qStep=c.auctionStep||10;
  if(auction.kind==='building'){
    const t=auction.targets?.[c.actor];
    if(!Number.isInteger(t))return 0;
    const bCost=BOARD[t]?.build||1000;
    const lv=c.view.assets[t]?.level||0;
    const mult=lv===2?2.25:lv===1?2.05:lv===3?1.85:1.55;
    const rawCap=Math.min(Math.max(0,c.self.cash-c.trueReserve*0.35),bCost*mult);
    return Math.floor(rawCap/qStep)*qStep;
  }
  const t=auction.tile,p=BOARD[t];
  if(!p)return 0;
  const comp=completion(c,t,c.actor);
  const blk=block(c,t);
  const tier=p.group>=0?(GROUP_TIER[p.group]||1):1;

  let maxVal=p.price*0.98;
  if(comp>=0.95){
    maxVal=p.price*Math.min(1.58,Math.max(1.32,tier*1.14))+360;
  }else if(blk>=0.80||c.boardGraph?.unownedCriticalDeny?.has(t)){
    const needyRival=c.groups[p.group]?.owners?.find(o=>o.count===GROUPS[p.group].ids.length-1);
    const rivalStillBidding=needyRival&&auction.participants?.includes(needyRival.id)&&!auction.passed?.includes(needyRival.id);
    const rivalCash=needyRival?c.view.players[needyRival.id]?.cash||0:0;
    const driveCap=p.price*1.36+240;
    maxVal=rivalStillBidding?Math.min(rivalCash+qStep,driveCap):p.price*1.12;
  }else if(c.boardGraph?.brokerTargetTiles?.has(t)){
    // 2-Hop Broker Bait in Auction: Winning this tile lets us trade it for our monopoly!
    maxVal=p.price*Math.min(1.22,Math.max(1.08,tier*0.96));
  }else if(c.earlyCashSinkTiles?.has(t)){
    maxVal=p.price*0.82;
  }else if(c.boardGraph?.poisonPillTiles?.has(t)){
    maxVal=p.price*Math.min(1.10,Math.max(0.98,tier*0.86));
  }else if(c.chainTargetTiles?.has(t)){
    maxVal=p.price*1.08;
  }else if(comp>=0.40){
    maxVal=p.price*Math.min(1.06,Math.max(0.98,tier*0.88));
  }else if(p.type==='station'){
    const ownSt=[5,15,25,35].filter(i=>owner(c,i)===c.actor&&i!==t).length;
    const geoMult=STATION_GEOMETRY_MULT[t]||1.0;
    maxVal=p.price*(0.96+ownSt*0.10)*geoMult;
  }else if(p.type==='utility'){
    maxVal=p.price*0.80;
  }

  const qReserve=(comp<0.90&&blk<0.80)?((c.queueAuctionReserve||0)+(c.vultureReserve||0)*0.5):0;
  const safeCash=Math.max(0,c.self.cash+(c.liquidBuffer||0)*0.35-c.trueReserve*0.45-qReserve);
  const rawLimit=Math.min(c.self.cash,safeCash,maxVal);
  return Math.floor(rawLimit/qStep)*qStep;
}

export function normalizeTrade(c,a){
  if(!['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(a.type))return null;
  const raw=(a.type==='TRADE_ACCEPT'||a.type==='TRADE_REJECT')?c.view.trade:a.trade;
  if(!raw)return null;
  const isIncoming=a.type==='TRADE_ACCEPT'||a.type==='TRADE_REJECT';
  const proposerId=isIncoming?raw.from:c.actor;
  const partnerId=isIncoming?raw.from:raw.to;
  const myGive=isIncoming?(raw.take||[]):(raw.give||[]);
  const myTake=isIncoming?(raw.give||[]):(raw.take||[]);
  const cashReceived=isIncoming?(raw.giveCash||0):(raw.takeCash||0);
  const cashPaid=isIncoming?(raw.takeCash||0):(raw.giveCash||0);
  const cardsReceived=isIncoming?(raw.giveCards||0):(raw.takeCards||0);
  const cardsPaid=isIncoming?(raw.takeCards||0):(raw.giveCards||0);
  const mortgageFeePaid=myTake.reduce((n,t)=>n+(c.view.assets[t]?.mortgaged?interest(t):0),0);
  const mortgageFeePartner=myGive.reduce((n,t)=>n+(c.view.assets[t]?.mortgaged?interest(t):0),0);
  return {
    raw,isIncoming,isReject:a.type==='TRADE_REJECT',
    proposerId,partnerId,myGive,myTake,
    netCash:cashReceived-cashPaid-mortgageFeePaid,
    partnerNetCash:cashPaid-cashReceived-mortgageFeePartner,
    netCards:cardsReceived-cardsPaid
  };
}

export function completesGroupPostTrade(c,playerId,acquired,surrendered){
  const acqSet=new Set(acquired),surSet=new Set(surrendered);
  let count=0,bestTier=0,buildCost3=0,groupIds=[];
  for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
    const ids=GROUPS[gIdx].ids;
    const hadBefore=ids.every(i=>owner(c,i)===playerId);
    const hasAfter=ids.every(i=>(owner(c,i)===playerId&&!surSet.has(i))||acqSet.has(i));
    if(hasAfter&&!hadBefore){
      count++;
      groupIds.push(gIdx);
      bestTier=Math.max(bestTier,GROUP_TIER[gIdx]||1);
      buildCost3+=ids.length*3*(BOARD[ids[0]]?.build||1000);
    }
  }
  return {count,bestTier,buildCost3,groupIds};
}

export function getBilateralTradeEvaluation(c,trade,proposerId=c.actor,stateOverride=null){
  const targetView=stateOverride||c.view;
  const partnerId=proposerId===c.actor?trade.to:proposerId;
  const key=(stateOverride?'override:':'live:')+proposerId+':'+partnerId+':'+JSON.stringify(trade);
  if(!stateOverride&&c._tradeEvalCache?.has(key))return c._tradeEvalCache.get(key);

  let partnerAnalysis=!stateOverride?c._partnerAnalyses?.get(partnerId):null;
  if(!partnerAnalysis){
    partnerAnalysis=analyzeState(targetView,partnerId,{fast:false});
    if(!stateOverride)c._partnerAnalyses?.set(partnerId,partnerAnalysis);
  }
  const proposerPublic=analyzeState(targetView,proposerId,{fast:true,publicOpponent:true});
  const partnerContext={
    observerId:partnerId,
    analyses:new Map([[partnerId,partnerAnalysis],[proposerId,proposerPublic]])
  };

  const myAnalysis=(!stateOverride&&c.ownAnalysis)||analyzeState(targetView,c.actor,{fast:false});
  const partnerPublic=analyzeState(targetView,partnerId,{fast:true,publicOpponent:true});
  const myContext={
    observerId:c.actor,
    analyses:new Map([[c.actor,myAnalysis],[partnerId,partnerPublic]])
  };

  const partnerEval=evaluateTrade(targetView,partnerId,trade,proposerId,partnerContext);
  const myEval=evaluateTrade(targetView,c.actor,trade,proposerId,myContext);
  const result={myEval,partnerEval};
  if(!stateOverride)c._tradeEvalCache?.set(key,result);
  return result;
}

export function predictPartnerAcceptance(c,trade,stateOverride=null){
  const targetView=stateOverride||c.view;
  const partnerId=trade.to;
  const partner=targetView.players[partnerId];
  if(!partner||partner.out)return {willAccept:false,partnerNet:-1e8,myEval:null,partnerEval:null};

  const {myEval,partnerEval}=getBilateralTradeEvaluation(c,trade,c.actor,targetView);
  if(partnerEval.invalid||partnerEval.insolventAfterFees||partnerEval.unilateralLoss||partnerEval.net<2){
    return {willAccept:false,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  const myMono=completesGroupPostTrade(c,c.actor,trade.take||[],trade.give||[]);
  const oppMono=completesGroupPostTrade(c,partnerId,trade.give||[],trade.take||[]);
  const persona=partner.botPersona||'human';

  if(persona==='kresna'){
    if(myMono.count>0&&oppMono.count===0){
      return {willAccept:false,partnerNet:partnerEval.net,myEval,partnerEval};
    }
    return {willAccept:partnerEval.net>=15,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  if(persona==='apex'){
    return {willAccept:partnerEval.net>=5,partnerNet:partnerEval.net,myEval,partnerEval};
  }

  const isCatchUp=c.catchUpPartners?.has(partnerId);
  const requiredNet=myMono.count>0&&oppMono.count===0?(isCatchUp?50:72):12;
  return {
    willAccept:partnerEval.net>=requiredNet,
    partnerNet:partnerEval.net,
    myEval,
    partnerEval
  };
}

export function solveOptimalTradeCash(c,{
  partnerId,giveList,takeList,giveCards=0,
  minOverpay=-360,maxOverpay=950,maxCashOffer,partnerMaxCash,stateOverride=null
}){
  const targetVal=takeList.reduce((n,id)=>n+(BOARD[id]?.price||0),0);
  const giveVal=giveList.reduce((n,id)=>n+(BOARD[id]?.price||0),0);
  const cardVal=giveCards*250;

  const makeOfferAt=(overpay)=>{
    const rawPay=targetVal-giveVal-cardVal+overpay;
    if(rawPay>maxCashOffer)return null;
    const giveCash=rawPay>0?Math.min(maxCashOffer,Math.round(rawPay/10)*10):0;
    const takeCash=rawPay<0?Math.min(partnerMaxCash,Math.round(Math.abs(rawPay)/10)*10):0;
    if(giveList.length===0&&giveCards===0&&giveCash<targetVal)return null;
    const trade={to:partnerId,give:giveList,take:takeList,giveCash,takeCash,giveCards,takeCards:0};
    const oracle=predictPartnerAcceptance(c,trade,stateOverride);
    const validForUs=oracle.willAccept&&oracle.myEval?.net>0&&(oracle.myEval.deltaShare>=oracle.myEval.partnerDelta-0.002);
    return {trade,oracle,validForUs,overpay};
  };

  const probePoints=[minOverpay,-180,0,120,260,420,600,780,maxOverpay].filter(x=>x>=minOverpay&&x<=maxOverpay);
  let highRes=null;
  let lowFail=minOverpay;

  for(const pt of probePoints){
    const res=makeOfferAt(pt);
    if(res?.validForUs){
      highRes=res;
      break;
    }
    lowFail=pt;
  }
  if(!highRes)return null;

  let lo=Math.floor(lowFail/10);
  let hi=Math.floor(highRes.overpay/10);
  let best=highRes;
  while(lo<=hi){
    const mid=Math.floor((lo+hi)/2);
    const cand=makeOfferAt(mid*10);
    if(cand?.validForUs){
      best=cand;
      hi=mid-1;
    }else{
      lo=mid+1;
    }
  }
  return best;
}

export function projectWinProbabilityDelta(c,a){
  const {view,actor,self,rivals,liquidBuffer,winProbability}=c;
  const alive=[self,...rivals].filter(p=>p&&!p.out);
  if(alive.length<=1)return {deltaWinProb:0,postWinProb:winProbability,leaderDelta:0};

  const cashDelta=new Map(alive.map(p=>[p.id,0]));
  const acqMap=new Map(alive.map(p=>[p.id,new Set()]));
  const surMap=new Map(alive.map(p=>[p.id,new Set()]));
  const buildBoost=new Map(alive.map(p=>[p.id,0]));

  if(a.type==='BUY'&&Number.isInteger(view.buy)){
    const t=BOARD[view.buy];
    if(t?.price){
      cashDelta.set(actor,-t.price);
      acqMap.get(actor)?.add(t.id);
    }
  }else if(a.type==='BID'&&view.phase==='auction'){
    const amt=a.amount||0;
    cashDelta.set(actor,-amt);
    if(view.auction?.kind==='building'){
      buildBoost.set(actor,0.65);
    }else if(Number.isInteger(view.auction?.tile)){
      acqMap.get(actor)?.add(view.auction.tile);
    }
  }else if(a.type==='BUILD'||a.type==='BUILD_JOIN'){
    const bCost=BOARD[a.tile]?.build||1000;
    cashDelta.set(actor,-bCost);
    buildBoost.set(actor,0.68);
  }else if(a.type==='REDEEM'||a.type==='MORTGAGE_REDEEM'){
    const t=a.type==='MORTGAGE_REDEEM'?view.mortgageDecision?.tile:a.tile;
    if(Number.isInteger(t)){
      cashDelta.set(actor,-redemption(t));
      const g=BOARD[t]?.group;
      if(g>=0&&c.groups[g]?.fullSetOwned)buildBoost.set(actor,0.55);
    }
  }else if(['OFFER','COUNTER','TRADE_ACCEPT'].includes(a.type)){
    const norm=normalizeTrade(c,a);
    if(norm&&!norm.isReject){
      cashDelta.set(actor,norm.netCash);
      cashDelta.set(norm.partnerId,norm.partnerNetCash);
      for(const tid of norm.myTake){
        acqMap.get(actor)?.add(tid);
        surMap.get(norm.partnerId)?.add(tid);
      }
      for(const tid of norm.myGive){
        surMap.get(actor)?.add(tid);
        acqMap.get(norm.partnerId)?.add(tid);
      }
    }
  }

  const logitPost=(pl)=>{
    const pid=pl.id;
    const acq=acqMap.get(pid)||new Set();
    const sur=surMap.get(pid)||new Set();
    const pCash=Math.max(100,pl.cash+(cashDelta.get(pid)||0));
    let propVal=0;
    for(const t of BOARD){
      if(!t.price)continue;
      const curOwner=view.assets[t.id]?.owner;
      const ownsAfter=(curOwner===pid&&!sur.has(t.id))||acq.has(t.id);
      if(ownsAfter){
        const mort=view.assets[t.id]?.mortgaged;
        const lv=view.assets[t.id]?.level||0;
        propVal+=(mort?t.mortgage:t.price)+(t.build||0)*lv*0.95;
        if(pid===actor){
          if(c.boardGraph?.brokerTargetTiles?.has(t.id))propVal+=t.price*0.42;
          if(c.boardGraph?.thirdPartyInterceptionTiles?.has(t.id))propVal+=t.price*0.48;
        }
      }
    }
    let pBuiltMonos=buildBoost.get(pid)||0;
    for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
      const g=GROUPS[gIdx];
      const tier=GROUP_TIER[gIdx]||1;
      const hasMono=g.ids.every(id=>{
        const curOwner=view.assets[id]?.owner;
        return (curOwner===pid&&!sur.has(id))||acq.has(id);
      });
      if(hasMono){
        const curMinLv=Math.min(...g.ids.map(id=>view.assets[id]?.level||0));
        const houseCost=BOARD[g.ids[0]]?.build||1000;
        const extraBuf=pid===actor?liquidBuffer*0.75:0;
        const potentialLv=Math.min(3,curMinLv+Math.floor(Math.max(0,pCash+extraBuf-500)/(houseCost*g.ids.length)));
        pBuiltMonos+=(1.1+potentialLv*0.62)*tier;
      }
    }
    return Math.log(Math.max(500,pCash+propVal))*1.18+pBuiltMonos*1.05;
  };

  const postLogits=Object.fromEntries(alive.map(pl=>[pl.id,logitPost(pl)]));
  const expSum=alive.reduce((s,pl)=>s+Math.exp(postLogits[pl.id]||0),0);
  const postWinProb=Math.exp(postLogits[actor]||0)/Math.max(1e-6,expSum);
  const leaderId=c.tableLeaderPlayer?.id;
  const leaderPostProb=(leaderId!=null&&leaderId!==actor)?Math.exp(postLogits[leaderId]||0)/Math.max(1e-6,expSum):0;

  return {
    deltaWinProb:postWinProb-winProbability,
    postWinProb,
    leaderPostProb
  };
}

export function evaluateTradeForActor(c,norm){
  if(!norm)return 0;
  const myPropGain=norm.myTake.reduce((n,t)=>n+valuation(c,t),0)-norm.myGive.reduce((n,t)=>n+valuation(c,t),0);
  const faceDelta=norm.myTake.reduce((n,t)=>n+BOARD[t].price,0)-norm.myGive.reduce((n,t)=>n+BOARD[t].price,0)+norm.netCash;
  const cardUnitVal=c.canGiveJailCardSweetener?180:420;
  const cashAndCardGain=norm.netCash+norm.netCards*cardUnitVal;
  const myMono=completesGroupPostTrade(c,c.actor,norm.myTake,norm.myGive);
  const oppMono=completesGroupPostTrade(c,norm.partnerId,norm.myGive,norm.myTake);

  const surrenderedBuffer=norm.myGive.reduce((n,t)=>n+(c.view.assets[t]?.mortgaged?0:(BOARD[t]?.mortgage||0)),0);
  const remainingLiquidBuffer=Math.max(0,(c.liquidBuffer||0)-surrenderedBuffer);
  const myPostCash=c.self.cash+norm.netCash+remainingLiquidBuffer*0.85;
  const oppPlayer=c.view.players[norm.partnerId];
  const oppPostCash=(oppPlayer?.cash||0)+norm.partnerNetCash;

  let strategicDelta=0;
  if(oppMono.count>0&&myMono.count===0){
    strategicDelta-=6800;
  }else if(myMono.count>0&&oppMono.count===0){
    const houseCost=BOARD[GROUPS[myMono.groupIds[0]].ids[0]]?.build||1000;
    const isDoubleSweep=norm.myTake.length>=2;
    const allowedFaceLoss=isDoubleSweep?-1550:-1150;
    if(myPostCash<houseCost*2||faceDelta<allowedFaceLoss){
      strategicDelta-=3000;
    }else{
      const buildReadiness=clamp(myPostCash/Math.max(1500,myMono.buildCost3*0.55),0.45,1.45);
      strategicDelta+=(isDoubleSweep?4400:3900)*myMono.bestTier*buildReadiness;
    }
  }else if(myMono.count>0&&oppMono.count>0){
    const myHouseCost=BOARD[GROUPS[myMono.groupIds[0]].ids[0]]?.build||1000;
    const oppHouseCost=BOARD[GROUPS[oppMono.groupIds[0]].ids[0]]?.build||1000;
    const myBuildableHouses=Math.min(9,Math.floor(Math.max(0,myPostCash-c.trueReserve*0.4)/myHouseCost));
    const oppBuildableHouses=Math.min(9,Math.floor(Math.max(0,oppPostCash-300)/oppHouseCost));
    const firstBuildAdvantage=!norm.isIncoming&&myBuildableHouses>=3;

    if(firstBuildAdvantage&&(myBuildableHouses>=oppBuildableHouses+2||oppBuildableHouses<=1||myMono.bestTier>oppMono.bestTier*1.12)){
      strategicDelta+=3700;
    }else if(norm.isIncoming&&oppBuildableHouses>=2){
      strategicDelta-=4600;
    }else{
      strategicDelta-=3900;
    }
  }else{
    if((c.view.phase==='debt'||(c.view.phase==='buy'&&c.needsPreBuyTrade)||(c.view.phase==='auction'&&c.needsMidAuctionTrade))&&norm.netCash>0&&faceDelta>=-320){
      strategicDelta+=2600;
    }else if(faceDelta<-260){
      strategicDelta-=1800;
    }else{
      for(const t of norm.myTake){
        if(c.boardGraph?.thirdPartyInterceptionTiles?.has(t))strategicDelta+=1250*(GROUP_TIER[BOARD[t]?.group]||1);
        if(c.boardGraph?.brokerTargetTiles?.has(t))strategicDelta+=1150*(GROUP_TIER[BOARD[t]?.group]||1);
        if(c.chainTargetTiles?.has(t))strategicDelta+=850*(GROUP_TIER[BOARD[t]?.group]||1);
        const b=BOARD[t];
        if(b?.group>=0&&c.groups[b.group].own===1&&GROUPS[b.group].ids.length===3){
          strategicDelta+=480*(GROUP_TIER[b.group]||1);
        }
      }
      if(c.needsBuildCapital&&norm.netCash>=900&&faceDelta>=-50&&norm.myGive.every(t=>!c.groups[BOARD[t]?.group]?.fullSetOwned)){
        strategicDelta+=780;
      }
    }
  }

  const rawScore=(myPropGain+cashAndCardGain+strategicDelta)/Math.max(1200,c.salary);
  const signal=clamp(rawScore,-1,1);
  return norm.isReject?clamp(-signal,-1,1):signal;
}

function sovereignScoreRebase(context,action,nativeScore){
  const {view,self,trueReserve,liquidBuffer}=context;
  const inflatedReserve=Math.max(trueReserve*1.6,self.cash*0.55);
  let spend=0;
  if(action.type==='BUILD'||action.type==='BUILD_JOIN'){
    if(context.shouldDeferPreRollBuild&&action.type==='BUILD')return nativeScore-450;
    spend=BOARD[action.tile]?.build||0;
  }else if(action.type==='BUY'){
    if(context.earlyCashSinkTiles?.has(view.buy))return nativeScore;
    spend=BOARD[view.buy]?.price||0;
  }else if(action.type==='REDEEM'||action.type==='MORTGAGE_REDEEM'){
    if(context.shouldDeferPreRollBuild&&action.type==='REDEEM')return nativeScore-380;
    const t=action.type==='MORTGAGE_REDEEM'?view.mortgageDecision?.tile:action.tile;
    const g=BOARD[t]?.group;
    if(Number.isInteger(t)&&((g>=0&&context.groups[g]?.fullSetOwned)||context.canDeleverage||action.type==='MORTGAGE_REDEEM')){
      spend=redemption(t);
    }
  }
  if(spend<=0)return nativeScore;
  const effectiveAfter=self.cash-spend+liquidBuffer*0.75;
  if(effectiveAfter<trueReserve)return nativeScore;

  const penaltyBefore=liquidityPenalty(self.cash,inflatedReserve);
  const penaltyAfter=liquidityPenalty(self.cash-spend,inflatedReserve);
  const artificialDrag=Math.max(0,penaltyAfter-penaltyBefore);
  return nativeScore+artificialDrag*0.86;
}

        if(d.group>=0&&GROUPS[d.group].ids.some(i=>(view.assets[i]?.level||0)>0))return false;
        const completesOurs=d.group>=0&&context.groups[d.group].own===GROUPS[d.group].ids.length-1;
        const isInterceptionOrBroker=boardGraph.thirdPartyInterceptionTiles.has(d.id)||boardGraph.brokerTargetTiles.has(d.id)||context.chainTargetTiles?.has(d.id);
        const isHighStationOrHotspot=d.id===5||(d.group>=0&&[1,3,4].includes(d.group));
        return completesOurs||isInterceptionOrBroker||isHighStationOrHotspot;
      }).sort((a,b)=>completion(context,b.id,actor)-completion(context,a.id,actor)||(GROUP_TIER[b.group]||1)-(GROUP_TIER[a.group]||1));

      const trojanBaits=(boardGraph.trojanBaitForRival.get(senderId)||[]).map(x=>BOARD[x.tileId]).filter(Boolean);

      let counterSynthesized=false;

      // TIER 1: Monopoly / Broker Hijack Counter using Trojan Baits
      for(const targetDeed of senderTargets.slice(0,2)){
        if(counterSynthesized)break;
        const giveOptions=trojanBaits.slice(0,2).map(b=>[b.id]);
        giveOptions.push([]);
        for(const giveList of giveOptions){
          if(counterSynthesized)break;
          const cardSweetener=(context.canGiveJailCardSweetener&&(self.cards?.length||0)>0)?1:0;
          const maxCash=Math.max(0,Math.floor((self.cash-context.trueReserve-900)/10)*10);
          const solved=solveOptimalTradeCash(context,{
            partnerId:senderId,
            giveList,
            takeList:[targetDeed.id],
            giveCards:cardSweetener,
            minOverpay:-200,
            maxOverpay:680,
            maxCashOffer:maxCash,
            partnerMaxCash:senderCash
          });
          if(solved){
            const cAct={type:'COUNTER',trade:solved.trade,actionId:'nusa-tier1-hijack-counter-'+targetDeed.id};
            const proj=projectWinProbabilityDelta(context,cAct);
            if(proj.deltaWinProb>0&&pushCandidate(cAct,topScore()+460)){
              counterSynthesized=true;
              break;
            }
          }
        }
      }

      // TIER 2: Extortionate Cash Toll Counter
      if(!counterSynthesized&&(t.take?.length||0)>0){
        const requestedSafe=t.take.every(tid=>{
          const b=BOARD[tid];
          if(!b||(view.assets[tid]?.level||0)>0)return false;
          if(b.group>=0&&context.groups[b.group].own>=2)return false;
          if(b.group>=0){
            const oppCnt=GROUPS[b.group].ids.filter(i=>view.assets[i]?.owner===senderId).length;
            if(oppCnt>=GROUPS[b.group].ids.length-1)return false;
          }
          return true;
        });
        if(requestedSafe&&(evalScore<0.12||acceptProj.deltaWinProb<=0)){
          const takeCandidates=senderTargets.length>0?[[senderTargets[0].id],[]]:[[]];
          for(const takeList of takeCandidates){
            if(counterSynthesized)break;
            const solvedToll=solveOptimalTradeCash(context,{
              partnerId:senderId,
              giveList:[...t.take],
              takeList,
              giveCards:0,
              minOverpay:-650,
              maxOverpay:-80,
              maxCashOffer:0,
              partnerMaxCash:senderCash
            });
            if(solvedToll&&solvedToll.trade.takeCash>0){
              const cAct={type:'COUNTER',trade:solvedToll.trade,actionId:'nusa-tier2-toll-counter-'+solvedToll.trade.takeCash};
              const proj=projectWinProbabilityDelta(context,cAct);
              if(proj.deltaWinProb>0&&pushCandidate(cAct,topScore()+395)){
                counterSynthesized=true;
                break;
              }
            }
          }
        }
      }

      // TIER 3: Toxic-Asset Swap Counter
      if(!counterSynthesized&&trojanBaits.length>0&&senderTargets.length>0){
        const toxicBait=trojanBaits.find(d=>d.type==='utility'||d.group===6)||trojanBaits[0];
        const primeTarget=senderTargets[0];
        const maxCash=Math.max(0,Math.floor((self.cash-context.trueReserve-700)/10)*10);
        const solvedSwap=solveOptimalTradeCash(context,{
          partnerId:senderId,
          giveList:[toxicBait.id],
          takeList:[primeTarget.id],
          giveCards:0,
          minOverpay:-250,
          maxOverpay:350,
          maxCashOffer:maxCash,
          partnerMaxCash:senderCash
        });
        if(solvedSwap){
          const cAct={type:'COUNTER',trade:solvedSwap.trade,actionId:'nusa-tier3-toxic-swap-'+primeTarget.id};
          const proj=projectWinProbabilityDelta(context,cAct);
          if(proj.deltaWinProb>0){
            pushCandidate(cAct,topScore()+365);
          }
        }
      }
    }
  }

  // 6. DEBT EMERGENCY RESCUE TRADE
  if(phase==='debt'&&view.debt?.from===actor&&self.cash<view.debt.amount&&canTrade(view,actor)&&context.roundTradeCount<2){
    const deficit=view.debt.amount-self.cash;
    if(context.liquidBuffer<deficit&&context.isolatedMortgageable.length>0){
      const mem=memoryFor(view,actor);
      const richRivals=context.rivals.filter(p=>p.cash>=400&&!rejectionActive(view,mem,p.id)).sort((a,b)=>b.cash-a.cash);
      let debtTradeSynthesized=false;
      for(const r of richRivals){
        if(debtTradeSynthesized)break;
        for(const deed of context.isolatedMortgageable){
          if(deed.group>=0){
            const oppCnt=GROUPS[deed.group].ids.filter(i=>view.assets[i]?.owner===r.id).length;
            if(oppCnt>=GROUPS[deed.group].ids.length-1)continue;
          }
          for(const mult of [1.0,0.92,0.85]){
            const askCash=Math.min(r.cash,Math.max(deed.mortgage+100,Math.floor((deed.price*mult)/10)*10));
            if(askCash<=deed.mortgage)continue;
            const trade={to:r.id,give:[deed.id],take:[],giveCash:0,takeCash:askCash,giveCards:0,takeCards:0};
            if(JSON.stringify(trade)===mem.lastSignature)continue;
            if(predictPartnerAcceptance(context,trade).willAccept){
              pushCandidate({type:'OFFER',trade,actionId:'nusa-debt-rescue-'+deed.id},topScore()+485);
              debtTradeSynthesized=true;
              break;
            }
          }
          if(debtTradeSynthesized)break;
        }
      }
    }
  }

  // 7. JAIL PHASE: Spatial Corridor Check
  if(phase==='jail'&&view.turn===actor&&context.unowned.length>=5&&context.nearLoss<420&&boardGraph.corridorRadar.spatialDangerFactor<1.15){
    if(self.cards?.length>0){
      pushCandidate({type:'JAIL_CARD',actionId:'nusa-early-jail-card'},topScore()+265);
    }else if(self.cash-500>=context.trueReserve+1200){
      pushCandidate({type:'JAIL_PAY',actionId:'nusa-early-jail-pay'},topScore()+245);
    }
  }

  if(phase==='roll'&&view.turn===actor&&context.shouldDeferPreRollBuild){
    pushCandidate({type:'ROLL',actionId:'nusa-disciplined-roll'},topScore()+(context.hasPendingRivalRentClaim?485:355));
  }

  // 8. POST-ROLL / JAIL DEVELOPMENT: Spatial-Corridor Prioritized Building & Macro Attachment
  if(['roll','end','jail'].includes(phase)&&view.turn===actor&&canManage(view,actor)&&!context.shouldDeferPreRollBuild){
    for(const g of context.groups){
      if(g.fullSetOwned&&g.mortgagedCount>0){
        const mTile=g.ids.find(id=>view.assets[id]?.mortgaged);
        if(mTile!==undefined){
          const rCost=redemption(mTile);
          if(self.cash-rCost>=context.trueReserve*0.45&&!managementError(view,actor,mTile,'REDEEM')){
            const redAct={type:'REDEEM',tile:mTile,actionId:'nusa-redeem-mono-'+mTile};
            const firstBuild={type:'BUILD',tile:g.apexTile};
            pushCandidate(redAct,topScore()+410,[redAct,firstBuild]);
          }else if(self.cash+context.liquidBuffer-rCost>=context.trueReserve*0.50){
            const coll=context.isolatedMortgageable.find(d=>!managementError(view,actor,d.id,'MORTGAGE'));
            if(coll){
              const mortAct={type:'MORTGAGE',tile:coll.id,actionId:'nusa-leverage-redeem-'+coll.id};
              pushCandidate(mortAct,topScore()+390,[mortAct,{type:'REDEEM',tile:mTile}]);
            }
          }
        }
      }
    }

    const maxTargetLevel=4;
    const buildGroups=context.groups
      .filter(g=>g.complete&&g.levelMin<maxTargetLevel)
      .sort((a,b)=>(b.ambushIntensity*1.8+b.tier)-(a.ambushIntensity*1.8+a.tier));

    for(const g of buildGroups){
      const levelMin=g.levelMin;
      const lockHouses=(context.sariPresent||context.rivalMonopolies>0||context.rivalHasHotel)?context.bank.houses<=18:context.bank.houses<=14;
      if(levelMin===4&&lockHouses)continue;

      const validTiles=g.ids
        .filter(id=>!managementError(view,actor,id,'BUILD'))
        .sort((x,y)=>{
          const apexX=APEX_TILES.has(x)?0.45:0;
          const apexY=APEX_TILES.has(y)?0.45:0;
          const ambX=boardGraph.corridorRadar.tileRivalRollProb[x]||0;
          const ambY=boardGraph.corridorRadar.tileRivalRollProb[y]||0;
          const scoreX=(context.tileKillshot[x]||0)*2.5+ambX*2.2+apexX;
          const scoreY=(context.tileKillshot[y]||0)*2.5+ambY*2.2+apexY;
          return scoreY-scoreX;
        });

      if(validTiles.length>0){
        const bestTile=validTiles[0];
        const bCost=BOARD[bestTile].build;
        const safetyFloor=(levelMin<3||context.hotelTrapActive)?context.trueReserve*0.42:context.trueReserve*0.78;
        if(self.cash-bCost>=safetyFloor){
          const ambushBoost=g.ambushIntensity>=0.20?25:0;
          const trapBoost=context.hotelTrapActive?420:(levelMin<3?400+ambushBoost:245+ambushBoost);
          const buildAct={type:'BUILD',tile:bestTile,actionId:'nusa-build-'+bestTile};
          const macroSeq=[buildAct];
          if(validTiles.length>1&&self.cash-bCost*2>=safetyFloor){
            macroSeq.push({type:'BUILD',tile:validTiles[1]});
          }
          pushCandidate(buildAct,topScore()+trapBoost,macroSeq);
        }else if(levelMin<3||context.hotelTrapActive){
          const collateral=context.isolatedMortgageable.find(t=>!managementError(view,actor,t.id,'MORTGAGE'));
          if(collateral&&self.cash+collateral.mortgage-bCost>=safetyFloor*0.85){
            const mortAct={type:'MORTGAGE',tile:collateral.id,actionId:'nusa-leverage-build-'+collateral.id};
            pushCandidate(mortAct,topScore()+370,[mortAct,{type:'BUILD',tile:bestTile}]);
          }
        }
      }
    }

    if(context.canDeleverage&&phase==='end'){
      const redeemTargets=context.mortgagedOwned
        .filter(t=>!context.cloakedMonopolyGroups?.has(t.group)&&!managementError(view,actor,t.id,'REDEEM')&&self.cash-redemption(t.id)>=context.trueReserve+900)
        .sort((a,b)=>(b.type==='station'?1:0)-(a.type==='station'?1:0)||b.price-a.price);
      if(redeemTargets.length>0){
        const rt=redeemTargets[0];
        pushCandidate({type:'REDEEM',tile:rt.id,actionId:'nusa-deleverage-'+rt.id},topScore()+225);
      }
    }
  }

  // 9. OMNISCIENT HYPERGRAPH TRADE SYNTHESIS (1-Hop Trojan Buyout, 2-Hop Broker, Interdiction, Double-Sweep)
  if(phase==='end'&&view.turn===actor&&canTrade(view,actor)&&context.roundTradeCount<3){
    for(const c of candidates){
      if(c.action.type==='OFFER'){
        const norm=normalizeTrade(context,c.action);
        const s=evaluateTradeForActor(context,norm);
        const oracle=predictPartnerAcceptance(context,c.action.trade);
        if(oracle.willAccept&&s>=0.22&&oracle.myEval.net>0&&oracle.myEval.deltaShare>=oracle.myEval.partnerDelta){
          c.score=Math.max(c.score+220,topScore()+200*s);
        }else if(!oracle.willAccept){
          c.score-=1500;
        }
      }
    }

    const mem=memoryFor(view,actor);
    let synthesizedCount=0;

    // A. 1-HOP DIRECT & TROJAN-BAIT MONOPOLY BUYOUT + 2-HOP BROKER ACQUISITION
    for(const bp of boardGraph.brokerPlans){
      if(synthesizedCount>=3)break;
      const {groupId:gId,tier,houseCost,missingTile,keyHolderId:targetOwner,directBaits,twoHopBaits}=bp;
      const g=context.groups[gId];
      if(g.ids.some(id=>(view.assets[id]?.level||0)>0))continue;

      let directSolved=false;
      if(!rejectionActive(view,mem,targetOwner)){
        const partnerMaxCash=view.players[targetOwner]?.cash||0;
        const isCatchUpPartner=context.catchUpPartners?.has(targetOwner);
        const packages=[];
        for(const b of directBaits.slice(0,3))packages.push([b.tileId]);
        if(directBaits.length>=2)packages.push([directBaits[0].tileId,directBaits[1].tileId]);
        packages.push([]);

        const cardOptions=(context.canGiveJailCardSweetener&&(self.cards?.length||0)>0)?[1,0]:[0];
        const maxOverpayCap=Math.round(780*tier+(isCatchUpPartner?180:0));

        for(const giveList of packages){
          if(directSolved)break;
          const surrenderedBuffer=giveList.reduce((n,id)=>n+(view.assets[id]?.mortgaged?0:(BOARD[id]?.mortgage||0)),0);
          const remainingBuffer=Math.max(0,context.liquidBuffer-surrenderedBuffer);
          const minBuildNeed=Math.max(0,houseCost*2+context.trueReserve*0.40-remainingBuffer*0.85);
          const maxCashOffer=Math.max(0,Math.floor((self.cash-minBuildNeed)/10)*10);

          for(const giveCards of cardOptions){
            if(directSolved)break;
            const solved=solveOptimalTradeCash(context,{
              partnerId:targetOwner,
              giveList,
              takeList:[missingTile],
              giveCards,
              minOverpay:isCatchUpPartner?-420:-300,
              maxOverpay:maxOverpayCap,
              maxCashOffer,
              partnerMaxCash
            });
            if(solved&&JSON.stringify(solved.trade)!==mem.lastSignature){
              const action={
                type:'OFFER',
                trade:solved.trade,
                actionId:'nusa-bsearch-buyout-'+missingTile+'-'+solved.trade.giveCash
              };
              const buildFollowUp={type:'BUILD',tile:g.apexTile};
              if(pushCandidate(action,topScore()+435+(giveCards*15)+(isCatchUpPartner?25:0),[action,buildFollowUp])){
                synthesizedCount++;
                directSolved=true;
                break;
              }
            }
          }
        }

        // Pre-Trade Liquidity Mortgage if direct buyout only failed due to cash cap
        if(!directSolved&&canManage(view,actor)&&context.isolatedMortgageable.length>0){
          const giveList=packages[0]||[];
          const nonBaitColl=context.isolatedMortgageable.find(d=>!giveList.includes(d.id)&&!managementError(view,actor,d.id,'MORTGAGE'));
          if(nonBaitColl){
            const boostedView={
              ...view,
              players:view.players.map((p,i)=>i===actor?{...p,cash:p.cash+nonBaitColl.mortgage}:p),
              assets:{...view.assets,[nonBaitColl.id]:{...view.assets[nonBaitColl.id],mortgaged:true}}
            };
            const boostedMaxCash=Math.max(0,Math.floor(((self.cash+nonBaitColl.mortgage)-houseCost*1.5)/10)*10);
            const solvedBoost=solveOptimalTradeCash(context,{
              partnerId:targetOwner,
              giveList,
              takeList:[missingTile],
              giveCards:0,
              minOverpay:0,
              maxOverpay:maxOverpayCap,
              maxCashOffer:boostedMaxCash,
              partnerMaxCash,
              stateOverride:boostedView
            });
            if(solvedBoost){
              const mortAct={type:'MORTGAGE',tile:nonBaitColl.id,actionId:'nusa-pretrade-mort-'+nonBaitColl.id};
              const offAct={type:'OFFER',trade:solvedBoost.trade};
              if(pushCandidate(mortAct,topScore()+422,[mortAct,offAct])){
                synthesizedCount++;
                directSolved=true;
              }
            }
          }
        }
      }

      // B. 2-HOP BROKER ACQUISITION ("Si A butuh Y, dan Y dipegang Si B -> Ambil Y dari Si B sekarang!")
      if(!directSolved&&twoHopBaits.length>0){
        for(const hop of twoHopBaits){
          if(synthesizedCount>=3)break;
          const partyB=hop.baitOwner;
          if(partyB==null||partyB===actor||rejectionActive(view,mem,partyB))continue;
          const baitsForB=(boardGraph.trojanBaitForRival.get(partyB)||[]).slice(0,2).map(x=>[x.tileId]);
          baitsForB.push([]);
          const maxCashB=Math.max(0,Math.floor((self.cash-context.trueReserve-houseCost*2)/10)*10);
          const partnerCashB=view.players[partyB]?.cash||0;

          let hopSolved=false;
          for(const giveListB of baitsForB){
            if(hopSolved)break;
            const solvedB=solveOptimalTradeCash(context,{
              partnerId:partyB,
              giveList:giveListB,
              takeList:[hop.baitTile],
              giveCards:0,
              minOverpay:-240,
              maxOverpay:320,
              maxCashOffer:maxCashB,
              partnerMaxCash:partnerCashB
            });
            if(solvedB&&JSON.stringify(solvedB.trade)!==mem.lastSignature){
              const step1Act={
                type:'OFFER',
                trade:solvedB.trade,
                actionId:'nusa-2hop-broker-step1-'+hop.baitTile+'-for-'+missingTile
              };
              const step2Trade={
                to:targetOwner,
                give:[hop.baitTile],
                take:[missingTile],
                giveCash:Math.max(0,Math.round(((BOARD[missingTile]?.price||0)-(BOARD[hop.baitTile]?.price||0)+180)/10)*10),
                takeCash:0,
                giveCards:0,
                takeCards:0
              };
              if(pushCandidate(step1Act,topScore()+415,[step1Act,{type:'OFFER',trade:step2Trade}])){
                synthesizedCount++;
                hopSolved=true;
                break;
              }
            }
          }
        }
      }
    }

    // C. PREEMPTIVE THIRD-PARTY MONOPOLY INTERDICTION ("Rebut kunci grup Si A dari tangan Si B!")
    if(synthesizedCount<3&&boardGraph.thirdPartyInterceptions.length>0){
      for(const inter of boardGraph.thirdPartyInterceptions){
        if(synthesizedCount>=3)break;
        const holderB=inter.holderRivalId;
        if(rejectionActive(view,mem,holderB))continue;
        const baitsForB=(boardGraph.trojanBaitForRival.get(holderB)||[])
          .filter(x=>BOARD[x.tileId]?.group!==inter.groupId)
          .slice(0,2)
          .map(x=>[x.tileId]);
        baitsForB.push([]);
        const maxCashB=Math.max(0,Math.floor((self.cash-context.trueReserve-900)/10)*10);
        const solvedInter=baitsForB.map(giveList=>solveOptimalTradeCash(context,{
          partnerId:holderB,
          giveList,
          takeList:[inter.tileId],
          giveCards:0,
          minOverpay:-220,
          maxOverpay:280,
          maxCashOffer:maxCashB,
          partnerMaxCash:view.players[holderB]?.cash||0
        })).find(Boolean);

        if(solvedInter&&JSON.stringify(solvedInter.trade)!==mem.lastSignature){
          const act={
            type:'OFFER',
            trade:solvedInter.trade,
            actionId:'nusa-interdict-snatch-'+inter.tileId
          };
          if(pushCandidate(act,topScore()+412)){
            synthesizedCount++;
          }
        }
      }
    }

    // D. 1-STEP DOUBLE-DEED MONOPOLY SWEEP (1/3 -> 3/3)
    if(synthesizedCount<3&&context.doubleSweepGroups.length>0){
      for(const ds of context.doubleSweepGroups){
        if(synthesizedCount>=3)break;
        const partnerId=ds.partnerId;
        if(rejectionActive(view,mem,partnerId))continue;
        const sweepBaits=(boardGraph.trojanBaitForRival.get(partnerId)||[])
          .filter(x=>BOARD[x.tileId]?.group!==ds.groupId);

        const sweepBundles=[];
        if(sweepBaits.length>=2)sweepBundles.push([sweepBaits[0].tileId,sweepBaits[1].tileId]);
        for(const b of sweepBaits.slice(0,2))sweepBundles.push([b.tileId]);

        const cardSweetener=(context.canGiveJailCardSweetener&&(self.cards?.length||0)>0)?1:0;
        for(const giveList of sweepBundles){
          const surrenderedBuffer=giveList.reduce((n,id)=>n+(view.assets[id]?.mortgaged?0:(BOARD[id]?.mortgage||0)),0);
          const remainingBuffer=Math.max(0,context.liquidBuffer-surrenderedBuffer);
          const minBuildRes=Math.max(0,ds.houseCost*2+context.trueReserve*0.40-remainingBuffer*0.85);
          const maxCashOffer=Math.max(0,Math.floor((self.cash-minBuildRes)/10)*10);
          const solvedSweep=solveOptimalTradeCash(context,{
            partnerId,
            giveList,
            takeList:ds.missingPair,
            giveCards:cardSweetener,
            minOverpay:-100,
            maxOverpay:900,
            maxCashOffer,
            partnerMaxCash:view.players[partnerId]?.cash||0
          });
          if(solvedSweep&&JSON.stringify(solvedSweep.trade)!==mem.lastSignature){
            const action={type:'OFFER',trade:solvedSweep.trade,actionId:'nusa-bsearch-sweep-'+ds.groupId};
            if(pushCandidate(action,topScore()+430)){
              synthesizedCount++;
              break;
            }
          }
        }
      }
    }
  }

  return plan;
}

function compatible(base,action,phase,context){
  if(eq(base,action))return true;
  if(protectedType(base)||protectedType(action))return false;
  if(phase==='buy'){
    const allowedBuy=context.needsPreBuyMortgage
      ?['BUY','AUCTION','MORTGAGE']
      :context.needsPreBuyTrade
        ?['BUY','AUCTION','OFFER']
        :['BUY','AUCTION'];
    return allowedBuy.includes(base.type)&&allowedBuy.includes(action.type);
  }
  if(phase==='auction'){
    const allowedAuction=context.needsMidAuctionMortgage
      ?['BID','PASS','MORTGAGE']
      :context.needsMidAuctionTrade
        ?['BID','PASS','OFFER']
        :['BID','PASS'];
    return allowedAuction.includes(base.type)&&allowedAuction.includes(action.type);
  }
  if(phase==='trade')return ['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(base.type)&&['OFFER','COUNTER','TRADE_ACCEPT','TRADE_REJECT'].includes(action.type);
  if(phase==='jail'){
    const allowedJail=['JAIL_PAY','JAIL_CARD','ROLL','BUILD','REDEEM'];
    return allowedJail.includes(base.type)&&allowedJail.includes(action.type);
  }
  if(phase==='buildOffer')return ['BUILD_JOIN','BUILD_DECLINE'].includes(base.type)&&['BUILD_JOIN','BUILD_DECLINE'].includes(action.type);
  if(phase==='mortgage')return ['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(base.type)&&['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(action.type);
  if(phase==='debt')return ['MORTGAGE','SELL','SELL_SET','OFFER'].includes(base.type)&&['MORTGAGE','SELL','SELL_SET','OFFER'].includes(action.type);

  if(phase==='roll'||phase==='end'){
    if(action.type==='SELL'||action.type==='SELL_SET')return false;
    if(action.type==='MORTGAGE'){
      const aid=String(action.actionId||'');
      const isSpecialMort=aid.startsWith('nusa-stealth-cloak-')||aid.startsWith('nusa-pretrade-mort-')||aid.startsWith('nusa-leverage-');
      const isIsolatedLeverage=context.isolatedMortgageable.some(d=>d.id===action.tile)&&context.needsBuildCapital&&!context.shouldDeferPreRollBuild;
      if(!isSpecialMort&&!isIsolatedLeverage)return false;
    }
    const allowedTurnTypes=phase==='roll'
      ?(context.shouldDeferPreRollBuild?['ROLL','BUILD','REDEEM']:['ROLL','BUILD','REDEEM','MORTGAGE'])
      :['END','BUILD','REDEEM','MORTGAGE','OFFER'];
    return allowedTurnTypes.includes(base.type)&&allowedTurnTypes.includes(action.type);
  }
  return base.type===action.type;
}

function computeAdvisory(signals,context,salary,action){
  if(!signals.length)return {advisory:0,bonus:0};
  let weightedSum=0,weightTotal=0;
  for(const s of signals){
    const w=weightFor(s.id,context);
    weightedSum+=s.signal*w;
    weightTotal+=w;
  }
  const norm=weightedSum/Math.max(1,Math.sqrt(weightTotal));
  let momentum=0;
  if(norm>0.20){
    const aid=String(action.actionId||'');
    if(action.type==='BUILD'||action.type==='BUILD_JOIN')momentum=salary*0.17;
    else if((action.type==='REDEEM'||action.type==='MORTGAGE_REDEEM')&&(context.hasMortgagedMonopoly||context.canDeleverage||context.view.phase==='mortgage'))momentum=salary*0.16;
    else if(action.type==='MORTGAGE'&&(context.needsPreBuyMortgage||context.needsMidAuctionMortgage||aid.startsWith('nusa-stealth-cloak-')||aid.startsWith('nusa-pretrade-mort-')||aid.startsWith('nusa-leverage-')))momentum=salary*0.19;
    else if(action.type==='OFFER'&&(norm>0.35||context.needsPreBuyTrade||context.needsMidAuctionTrade))momentum=salary*0.19;
    else if(action.type==='COUNTER'&&norm>0.35)momentum=salary*0.18;
    else if(action.type==='BUY'&&norm>0.25)momentum=salary*0.14;
    else if(action.type==='AUCTION'&&context.earlyCashSinkTiles?.has(context.view.buy))momentum=salary*0.16;
    else if(action.type==='ROLL'&&context.shouldDeferPreRollBuild)momentum=salary*0.18;
  }
  const bonus=Math.max(-salary*0.54,Math.min(salary*0.54,norm*salary*0.29+momentum));
  return {advisory:norm,bonus};
}

export function nusaV7Decision(state,actor,plan,{diagnostics=false,disabled=[]}={}){
  const fallback=(reason,extra={})=>({action:plan?.action??null,changed:false,reason,...extra});
  if(!plan?.action||protectedType(plan.action)||(plan.candidates?.[0]?.score??0)>=1e7)return fallback('protected_or_trivial');
  if(!['auction','buy','roll','end','trade','jail','buildOffer','mortgage','debt'].includes(state.phase))return fallback('protected_phase');

  const context=situationOf(state,actor,plan);
  ensureSovereignCandidates(context,plan);

  const base=plan.action,candidates=plan.candidates||[];
  if(candidates.length<2)return fallback('single_candidate');

  const salary=context.salary;
  const allowed=new Set(STRATEGY_EXPERTS.map(x=>x.id));
  if(disabled.some(id=>!allowed.has(id)))throw Error('Unknown NUSA expert ablation: '+disabled.find(id=>!allowed.has(id)));

  const current=ranked.find(c=>eq(c.action,base));
  if(!current)return fallback('baseline_not_ranked');
  const best=ranked[0];
  const improvement=best.unified-current.unified;

  const changeAllowed=!eq(best.action,base)&&
    improvement>Math.max(5,salary*0.002)&&
    best.score>=current.score-salary*0.92&&
    legalCandidate(context.view,actor,best.action);

  const result={
    action:changeAllowed?best.action:base,
    changed:changeAllowed,
    reason:changeAllowed?'omniscient_cortex_v22_selected':'native_plan_preserved',
    rankedPool:ranked
  };

  if(diagnostics){
    result.context={
      stage:context.stage,live:context.live,unowned:context.unowned.length,
      bankHouses:context.bank.houses,reserve:Math.round(context.reserve),
      trueReserve:Math.round(context.trueReserve),

      brokerPlans:context.boardGraph.brokerPlans.length,
      brokerTargets:context.boardGraph.brokerTargetTiles.size,
      thirdPartyInterceptions:context.boardGraph.thirdPartyInterceptions.length,
      hostageGroupsHeld:context.boardGraph.hostageGroupsHeld,
      effectiveTradeCash:context.effectiveTradeCash,canUsePovertyCloak:context.canUsePovertyCloak,
      auctionStep:context.auctionStep,hotelTrapActive:context.hotelTrapActive,

    };
    result.proposals=ranked.slice(0,8).map(c=>({id:c.id,type:c.action.type,score:+c.score.toFixed(2),unified:+c.unified.toFixed(2)}));
    result.expertCount=STRATEGY_EXPERTS.length;
    result.candidateCount=candidates.length;
    result.eligibleCount=eligible.length;
  }
  return result;
}

function comparableCandidate(first,cand,phase){
  if(eq(first,cand))return true;
  if(isBuild(first))return isBuild(cand);
  if(first.type==='OFFER')return cand.type==='OFFER'||isBuild(cand)||cand.type==='MORTGAGE'||(phase==='buy'&&cand.type==='AUCTION')||(phase==='auction'&&cand.type==='PASS');
  if(phase==='buy')return ['BUY','AUCTION','MORTGAGE','OFFER'].includes(cand.type);
  if(phase==='auction')return ['BID','PASS','MORTGAGE','OFFER'].includes(cand.type);
  if(phase==='buildOffer')return ['BUILD_JOIN','BUILD_DECLINE'].includes(cand.type);
  if(phase==='mortgage')return ['MORTGAGE_REDEEM','MORTGAGE_KEEP'].includes(cand.type);
  if(phase==='trade')return ['TRADE_ACCEPT','COUNTER','TRADE_REJECT'].includes(cand.type);
  if(phase==='jail')return ['JAIL_PAY','JAIL_CARD','ROLL','BUILD','REDEEM'].includes(cand.type);
  return first.type===cand.type;
}

function searchCandidatePool(plan,council,phase,maxRoots){
  const first=council.action;
  const source=council.rankedPool||plan.candidates;
  const selected=source.filter(c=>comparableCandidate(first,c.action,phase));
  const base=selected.find(c=>eq(c.action,first));
  if(!base)return [];
  const baseVal=base.unified??base.score;
  const window=Math.max(300,Math.abs(baseVal)*0.28);
  return [
    base,
    ...selected
      .filter(c=>c!==base&&(c.unified??c.score)>=baseVal-window)
      .sort((a,b)=>(b.unified??b.score)-(a.unified??a.score)||b.score-a.score||a.id.localeCompare(b.id))
  ].slice(0,maxRoots);
}

function sovereignLeafValue(s,actor){
  const posVal=positionValue(s,actor,{noRisk:false,noLearning:true});
  if(s.players[actor]?.out||s.phase==='won')return posVal;
  const myWorth=worth(s,actor);
  const alive=s.players.filter(p=>!p.out);
  const aliveOthers=alive.filter(p=>p.id!==actor);
  const maxRivalWorth=Math.max(1,...aliveOthers.map(p=>worth(s,p.id)));
  const worthEdge=(myWorth-maxRivalWorth*0.85)*0.26;

  let monopolyBonus=0;
  let hostageControlBonus=0;
  for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
    const g=GROUPS[gIdx];
    const tier=GROUP_TIER[gIdx]||1;
    if(complete(s,actor,gIdx)){
      const minLv=Math.min(...g.ids.map(i=>s.assets[i]?.level||0));
      monopolyBonus+=(490+Math.min(3,minLv)*310)*tier;
    }else{
      const myCnt=g.ids.filter(i=>s.assets[i]?.owner===actor).length;
      if(myCnt>=1){
        const rivalHasTwo=aliveOthers.some(r=>g.ids.filter(i=>s.assets[i]?.owner===r.id).length===g.ids.length-1);
        if(rivalHasTwo)hostageControlBonus+=240*tier;
        else if(myCnt===g.ids.length-1)hostageControlBonus+=160*tier;
      }
    }
  }

  const logitOf=(pl)=>{
    const w=Math.max(500,worth(s,pl.id));
    let mScore=0;
    for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
      if(complete(s,pl.id,gIdx)){
        const minLv=Math.min(...GROUPS[gIdx].ids.map(i=>s.assets[i]?.level||0));
        mScore+=1.1+Math.min(3,minLv)*0.62;
      }
    }
    return Math.log(w)*1.18+mScore*1.05;
  };
  const logits=Object.fromEntries(alive.map(pl=>[pl.id,logitOf(pl)]));
  const expSum=alive.reduce((acc,pl)=>acc+Math.exp(logits[pl.id]||0),0);
  const winProb=Math.exp(logits[actor]||0)/Math.max(1e-6,expSum);
  const winProbBonus=winProb*1680;

  return posVal*0.68+worthEdge+monopolyBonus+hostageControlBonus+winProbBonus;
}

function fastStrategicRollout(s,who,rng,rootActor,auctionQuantum){
  const p=s.players[who];
  if(!p||p.out)return null;

  if((s.rentClaims||[]).some(c=>c.to===who)){
    return {type:'CLAIM_RENT'};
  }

  const phase=s.phase;
  if(phase==='starting')return {type:'START_ROLL'};
  if(phase==='card')return {type:'ACK'};

  if(phase==='mortgage'){
    const mTile=s.mortgageDecision?.tile;
    if(Number.isInteger(mTile)){
      const rCost=redemption(mTile);
      const g=BOARD[mTile]?.group;
      const inFull=g>=0&&GROUPS[g].ids.every(i=>s.assets[i]?.owner===who);
      if(p.cash>=rCost&&(inFull||p.cash-rCost>=800)){
        return {type:'MORTGAGE_REDEEM'};
      }
    }
    return {type:'MORTGAGE_KEEP'};
  }

  if(phase==='buildOffer'){
    if(s.buildOffer?.pending?.includes(who)){
      for(const g of GROUPS){
        for(let idx=g.ids.length-1;idx>=0;idx--){
          const tid=g.ids[idx];
          if(!managementError(s,who,tid,'BUILD'))return {type:'BUILD_JOIN',tile:tid};
        }
      }
    }
    return {type:'BUILD_DECLINE'};
  }

  if(phase==='jail'){
    const unownedCount=BOARD.filter(t=>t.price&&s.assets[t.id]?.owner==null).length;
    if(unownedCount>=5){
      if(p.cards?.length>0)return {type:'JAIL_CARD'};
      if(p.cash>=1400)return {type:'JAIL_PAY'};
    }
    return {type:'ROLL'};
  }

  if(phase==='roll'){
    return {type:'ROLL'};
  }

  if(phase==='buy'){
    const t=BOARD[s.buy];
    const price=t?.price||0;
    if(who===rootActor){
      const isCashSink=t?.group===6&&GROUPS[6].ids.every(i=>s.assets[i]?.owner==null)&&(p.cash-price<2600);
      if(!isCashSink&&p.cash>=price)return {type:'BUY'};
      if(!isCashSink&&p.cash<price){
        const coll=BOARD.find(d=>d.price&&d.id!==s.buy&&s.assets[d.id]?.owner===who&&!s.assets[d.id]?.mortgaged&&(s.assets[d.id]?.level||0)===0&&!managementError(s,who,d.id,'MORTGAGE'));
        if(coll&&p.cash+coll.mortgage>=price)return {type:'MORTGAGE',tile:coll.id};
      }
      return {type:'AUCTION'};
    }
    const persona=p.botPersona||'human';
    const reserve=persona==='sari'?900:persona==='bima'?300:500;
    if(p.cash>=price+reserve||(p.cash>=price&&t?.group>=0&&GROUPS[t.group].ids.some(i=>s.assets[i]?.owner===who))){
      return {type:'BUY'};
    }
    return {type:'AUCTION'};
  }

  if(phase==='auction'){
    const qStep=auctionQuantum?.step||10;
    const curAmt=s.auction?.amount||0;
    const minBid=curAmt===0?(auctionQuantum?.minBid||100):Math.ceil((curAmt+qStep)/qStep)*qStep;
    if(minBid>p.cash)return {type:'PASS'};

    if(s.auction?.kind==='building'){
      const tId=s.auction.targets?.[who];
      const bCost=BOARD[tId]?.build||1000;
      const ceiling=Math.floor((bCost*(who===rootActor?2.05:1.65))/qStep)*qStep;
      return (minBid<=ceiling&&minBid<=p.cash-450)?{type:'BID',amount:minBid}:{type:'PASS'};
    }

    const t=BOARD[s.auction?.tile];
    if(!t?.price)return {type:'PASS'};
    const completes=t.group>=0&&GROUPS[t.group].ids.filter(i=>s.assets[i]?.owner===who&&i!==t.id).length===GROUPS[t.group].ids.length-1;
    if(who===rootActor){
      const isCashSink=t.group===6||t.type==='utility';
      const geoMult=t.type==='station'?(STATION_GEOMETRY_MULT[t.id]||1):1;
      const ceiling=completes
        ?Math.floor((t.price*1.42)/qStep)*qStep
        :isCashSink
          ?Math.floor((t.price*0.82)/qStep)*qStep
          :Math.floor((t.price*0.98*geoMult)/qStep)*qStep;
      if(minBid<=ceiling&&minBid<=p.cash-500){
        const stepAdd=ceiling-minBid>250?qStep*2:qStep;
        const bidAmt=Math.min(ceiling,Math.floor((curAmt+stepAdd)/qStep)*qStep);
        return {type:'BID',amount:Math.max(minBid,bidAmt)};
      }
      return {type:'PASS'};
    }
    const persona=p.botPersona||'human';
    const mult=completes?1.32:(persona==='kresna'||persona==='bima'?1.15:persona==='raka'&&t.type==='station'?1.22:1.0);
    const limit=Math.min(p.cash*0.76,t.price*mult);
    return minBid<=limit?{type:'BID',amount:minBid}:{type:'PASS'};
  }

  if(phase==='trade'&&s.trade){
    if(who!==rootActor){
      const ev=evaluateTrade(s,who,s.trade,s.trade.from);
      return (ev.net>=0&&!ev.invalid&&!ev.insolventAfterFees)?{type:'TRADE_ACCEPT'}:{type:'TRADE_REJECT'};
    }
    return {type:'TRADE_REJECT'};
  }

  if(phase==='debt'){
    if(p.cash>=(s.debt?.amount||0))return {type:'PAY'};
    for(const t of BOARD){
      if(t.price&&s.assets[t.id]?.owner===who&&!s.assets[t.id]?.mortgaged&&(s.assets[t.id]?.level||0)===0&&!managementError(s,who,t.id,'MORTGAGE')){
        return {type:'MORTGAGE',tile:t.id};
      }
    }
    for(const t of BOARD){
      if(t.price&&s.assets[t.id]?.owner===who&&(s.assets[t.id]?.level||0)>0){
        if(!managementError(s,who,t.id,'SELL'))return {type:'SELL',tile:t.id};
        if(!managementError(s,who,t.id,'SELL_SET'))return {type:'SELL_SET',tile:t.id};
      }
    }
    return {type:'BANKRUPT'};
  }

  if(phase==='end'){
    for(const g of GROUPS){
      if(g.ids.every(i=>s.assets[i]?.owner===who)){
        const mTile=g.ids.find(i=>s.assets[i]?.mortgaged);
        if(mTile!==undefined){
          const rCost=redemption(mTile);
          if(p.cash-rCost>=450&&!managementError(s,who,mTile,'REDEEM')){
            return {type:'REDEEM',tile:mTile};
          }
        }
      }
    }
    for(let gIdx=0;gIdx<GROUPS.length;gIdx++){
      const g=GROUPS[gIdx];
      if(g.ids.every(i=>s.assets[i]?.owner===who&&!s.assets[i]?.mortgaged)){
        const minLv=Math.min(...g.ids.map(i=>s.assets[i]?.level||0));
        if(minLv<4){
          for(let idx=g.ids.length-1;idx>=0;idx--){
            const tid=g.ids[idx];
            if((s.assets[tid]?.level||0)===minLv&&!managementError(s,who,tid,'BUILD')){
              const bCost=BOARD[tid]?.build||1000;
              if(p.cash-bCost>=550){
                return {type:'BUILD',tile:tid};
              }else if(who===rootActor&&minLv<3){
                const coll=BOARD.find(d=>d.price&&(d.group!==gIdx)&&s.assets[d.id]?.owner===who&&!s.assets[d.id]?.mortgaged&&(s.assets[d.id]?.level||0)===0&&(d.group<0||!GROUPS[d.group].ids.every(k=>s.assets[k]?.owner===who))&&!managementError(s,who,d.id,'MORTGAGE'));
                if(coll&&p.cash+coll.mortgage-bCost>=450){
                  return {type:'MORTGAGE',tile:coll.id};
                }
              }
            }
          }
        }
      }
    }
    return {type:'END'};
  }

  return rolloutPolicy(s,who,rng,{rootActor});
}

function evaluateWorld(publicView,actor,candidate,scenarioSeed,opts,expired,auctionQuantum){
  const random=new ScenarioRandom(scenarioSeed);
  let s=sampleWorld(publicView,random.stream(0,0,'sampled-decks'));
  const initial=sovereignLeafValue(s,actor);
  let nodes=0;
  try{
    const root=command(s,actor,candidate.action,random.forAction(s,actor,candidate.action));
    nodes++;s=root;
    if(s.log?.length>80)s.log=s.log.slice(-80);
    const path=Array.isArray(candidate.macro)?candidate.macro.slice(1,6):[];
    for(const part of path){
      if(expired()||simulationActor(s)!==actor||!legalCandidate(s,actor,part))break;
      s=command(s,actor,part,random.forAction(s,actor,part));nodes++;
    }
    const horizon=publicView.turnSerial+opts.horizonTurns;
    const maxEngineSteps=Math.max(opts.horizonSteps,opts.horizonTurns*12);
    for(let step=0;step<maxEngineSteps&&s.phase!=='won';step++){
      if(s.turnSerial>=horizon&&['roll','jail','won'].includes(s.phase))break;
      if(expired())return {ok:false,nodes,reason:'budget'};
      const who=simulationActor(s);
      const action=fastStrategicRollout(s,who,random.policy(s,who),actor,auctionQuantum);
      if(!action)break;
      try{
        s=command(s,who,action,random.forAction(s,who,action));
      }catch{
        const fallbackAct=rolloutPolicy(s,who,random.policy(s,who),{rootActor:actor});
        if(!fallbackAct)break;
        s=command(s,who,fallbackAct,random.forAction(s,who,fallbackAct));
      }
      nodes++;
      if(s.log?.length>80)s.log=s.log.slice(-80);
    }
    return {ok:true,nodes,value:sovereignLeafValue(s,actor)-initial};
  }catch{
    return {ok:false,nodes,reason:'sandbox_exception'};
  }
}

export function nusaV71Decision(state,actor,rawPlan,options={}){
  const began=performance.now();
  const phase=state.phase;
  const plan=expandNusaCandidates(state,actor,rawPlan,{maxExtra:6});
  const council=nusaV7Decision(state,actor,plan,{disabled:options.disabled||[],diagnostics:!!options.diagnostics});
  const result=(reason,extra={})=>({
    action:council.action,
    changed:council.changed,
    reason,
    search:{attempted:false,gear:1,pairedSamples:0,nodes:0,wallMs:+(performance.now()-began).toFixed(2),reason},
    ...extra
  });
  if(options.disableSearch)return result('search_disabled');
  if(!council.action||guardedPhase.has(phase)||protectedTypes.has(council.action.type)||
     !safeTypes.has(council.action.type))return result('gear1_instant_reflex');

  const requestedMs=Number(options.maxMs??options.timeBudgetMs??64);
  const deepMode=options.deep===true||requestedMs>=1500;
  const gear=deepMode?3:2;

  const maxMs=gear===3?Math.min(4200,Math.max(1800,requestedMs)):Math.min(240,Math.max(24,requestedMs));
  const maxNodes=gear===3?Math.min(9600,Math.max(1800,options.maxNodes??5200)):Math.min(1200,Math.max(160,options.maxNodes??540));
  const maxRoots=gear===3?Math.min(5,Math.max(2,options.maxRoots??4)):Math.min(4,Math.max(2,options.maxRoots??3));
  const maxSamples=gear===3?Math.min(32,Math.max(6,options.maxSamples??24)):Math.min(14,Math.max(4,options.maxSamples??8));
  const horizonTurns=gear===3?14:6;
  const horizonSteps=gear===3?112:48;

  const pool=searchCandidatePool(plan,council,phase,maxRoots);
  if(pool.length<2)return result('gear1_single_dominant_candidate');
  const view=observation(state,actor);
  if(pool.some(c=>!legalCandidate(view,actor,c.action)))return result('illegal_candidate_rejected');

  const auctionQuantum=getAuctionQuantum(view,actor);
  const expired=()=>performance.now()-began>=maxMs;
  const estimatedSeed=hashSeed(state.matchId+':'+state.version+':'+actor+':'+NUSA_REASONING_VERSION);
  const returns=pool.map(()=>[]);
  let activeIndices=pool.map((_,i)=>i);
  let nodes=0,samples=0,reason='completed';

  const halvingPoint=Math.max(3,Math.floor(maxSamples*0.45));

  for(let batch=0;batch<maxSamples;batch++){
    if(expired()||nodes>=maxNodes){reason='budget_exhausted';break;}
    const batchValues=new Map();
    for(const idx of activeIndices){
      if(expired()||nodes>=maxNodes){reason='budget_exhausted';break;}
      const root=pool[idx];
      const seed=hashSeed(estimatedSeed+':batch:'+batch);
      const evaluation=evaluateWorld(view,actor,root,seed,{horizonTurns,horizonSteps},expired,auctionQuantum);
      nodes+=evaluation.nodes;
      if(!evaluation.ok){reason=evaluation.reason||'simulation_failed';break;}
      batchValues.set(idx,evaluation.value);
    }
    if(batchValues.size!==activeIndices.length)break;
    for(const [idx,val] of batchValues.entries()){
      returns[idx].push(val);
    }
    samples++;

    if(samples===halvingPoint&&activeIndices.length>2){
      const scored=activeIndices.map(idx=>{
        const rs=returns[idx];
        const mean=rs.reduce((a,b)=>a+b,0)/rs.length;
        const baseU=pool[idx].unified??pool[idx].score;
        return {idx,u:baseU+mean*0.35};
      }).sort((a,b)=>b.u-a.u);
      const keepCount=Math.max(2,Math.ceil(activeIndices.length/2));
      const kept=new Set([0,...scored.slice(0,keepCount).map(x=>x.idx)]);
      activeIndices=[...kept].sort((a,b)=>a-b);
    }
  }

  const wallMs=+(performance.now()-began).toFixed(2);
  const search={attempted:true,gear,pairedSamples:samples,nodes,wallMs,reason,roots:pool.length};
  if(samples<2)return {...result(reason),search};

  const evidence=choosePairedSearchEvidence(pool,returns,view.rules?.salary??2000);
  const enough=evidence.selectedIndex>0;
  return {
    action:enough?pool[evidence.selectedIndex].action:council.action,
    changed:enough||council.changed,
    reason:enough?('omniscient_cortex_gear'+gear+'_crn_selected'):council.reason,
    search
  };
}

export function choosePairedSearchEvidence(pool,returns,salary){
  if(pool.length<2||returns.length!==pool.length||!returns[0]?.length)
    return {selectedIndex:0,bestDelta:0};
  const baseScore=pool[0].unified??pool[0].score;
  const utilities=pool.map((c,i)=>{
    const rs=returns[i];
    if(!rs||!rs.length)return -Infinity;
    const mean=rs.reduce((a,b)=>a+b,0)/rs.length;
    const sorted=rs.slice().sort((a,b)=>a-b);
    const tailCount=Math.max(1,Math.ceil(sorted.length*0.25));
    const cvar=sorted.slice(0,tailCount).reduce((a,b)=>a+b,0)/tailCount;
    const blended=mean*0.72+cvar*0.28;
    return ((c.unified??c.score)-baseScore)+blended*0.38;
  });
  const order=pool.map((_,i)=>i).sort((a,b)=>utilities[b]-utilities[a]);
  const best=order[0],delta=utilities[best]-utilities[0];
  const bestReturns=returns[best],ctrlReturns=returns[0];
  const commonLen=Math.min(bestReturns.length,ctrlReturns.length);
  const pairs=bestReturns.slice(0,commonLen).map((v,i)=>v-ctrlReturns[i]);
  const wins=pairs.filter(d=>d>0).length;
  const enough=best!==0&&commonLen>=2&&wins>=Math.ceil(commonLen*0.65)&&delta>Math.max(25,salary*0.012);
  return {selectedIndex:enough?best:0,bestDelta:+delta.toFixed(2)};
}
`;
